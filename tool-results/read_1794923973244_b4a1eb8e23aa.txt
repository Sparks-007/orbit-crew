    91→type ApprovalRow = {
    92→  id: string;

   100→type ChoiceAnswerRow = { answer: unknown; text: string };
   101→
   102→function approvalState(row: ApprovalRow): ApprovalState {
   103→  if (row.status === 'pending') return row.active ? 'pending' : 'expired';
   104→  if (row.status === 'confirmed') return row.executionActive ? 'working' : 'expired';
   105→  if (row.status === 'executing') return row.executionActive ? 'working' : 'uncertain';
   106→  if (row.status === 'completed') return 'done';
   107→  if (row.status === 'declined' || row.status === 'uncertain') return row.status;
   108→  return 'expired';
   109→}
   110→
   111→function choiceAnswerView(options: AnswerChoice[], row: ChoiceAnswerRow | null | undefined) {
   112→  const answer = questionAnswerSchema.safeParse(row?.answer);
   113→  if (!row || !answer.success) return undefined;
   114→  if (!('choiceIds' in answer.data)) return { choiceIds: [], text: row.text };
   115→  const { choiceIds, custom } = answer.data;
   116→  const text =
   117→    custom &&
   118→    writtenAnswer(
   119→      options.filter((option) => choiceIds.includes(option.id)),
   120→      row.text,
   121→    );
   122→  return { choiceIds, ...(text ? { text } : {}) };
   123→}

   125→type MessageRow = Omit<Message, 'approval' | 'choiceAnswer' | 'cardAnswer' | 'work'> & {
   126→  work?: Message['work'] | null;
   127→  assistantReaction?: MessageReaction | null;
   128→  approval?: ApprovalRow | null;
   129→  choiceAnswer?: ChoiceAnswerRow | null;
   130→  cardAnswer?: boolean | null;
   131→};

   133→function messageView(row: MessageRow): Message {
   134→  const {
   135→    assistantReaction,
   136→    presentation,
   137→    displayText,
   138→    audioCaption,
   139→    answerChoices,
   140→    answerChoicesPending,
   141→    approval,
   142→    choiceAnswer,
   143→    cardAnswer,
   144→    work,
   145→    ...message
   146→  } = row;
   147→  const visual =
   148→    message.role === 'assistant' ? presentationSchema.safeParse(presentation) : undefined;
   149→  const choices =
   150→    message.role === 'assistant' ? answerChoicesSchema.safeParse(answerChoices) : undefined;
   151→  const answered = choices?.success
   152→    ? choiceAnswerView(choices.data.options, choiceAnswer)
   153→    : undefined;
   154→  return {
   155→    ...message,
   156→    ...(message.role === 'user' && typeof displayText === 'string' ? { displayText } : {}),
   157→    ...(typeof audioCaption === 'string' ? { audioCaption } : {}),
   158→    ...(visual?.success ? { presentation: visual.data } : {}),
   159→    ...(choices?.success
   160→      ? { answerChoices: choices.data, answerChoicesPending: answerChoicesPending === true }
   161→      : {}),
   162→    ...(answered ? { choiceAnswer: answered } : {}),
   163→    ...(approval
   164→      ? {
   165→          approval: {
   166→            id: approval.id,
   167→            state: approvalState(approval),
   168→            ...approvalTitle(approval.action),
   169→            ...(approval.title ? { title: approval.title } : {}),
   170→            ...approvalContent(approval.action, approval.args),
   171→          },
   172→        }
   173→      : {}),
   174→    ...(cardAnswer ? { cardAnswer: true as const } : {}),
   175→    ...(work ? { work } : {}),
   176→    reactions: [
   177→      ...(message.reaction ? [{ actor: 'user' as const, reaction: message.reaction }] : []),
   178→      ...(assistantReaction ? [{ actor: 'assistant' as const, reaction: assistantReaction }] : []),
   179→    ],
   180→  };
   181→}

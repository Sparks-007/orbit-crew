   269→  async function everyFrame(
   270→    expression: string,
   271→    shadows?: 'document' | 'focus',
   272→    strict = Boolean(shadows),
   273→  ) {
   274→    const results: Array<FocusedField & { value: unknown }> = [];

   276→      const frames = await tree(session).catch((error) => {

   290→          root: DomNode;
   291→        };

   293→      }

   295→        try {
   296→          const context = await world(session, frame.id);
   297→          if (shadows === 'focus') {
   298→            const { result } = (await send(
   299→              'Runtime.evaluate',
   300→              {
   301→                contextId: context,
   302→                expression: `globalThis.__nodRoots = []; ${active}`,
   303→                objectGroup: 'nod-roots',
   304→              },
   305→              session,
   306→            )) as { result?: { objectId?: string } };

   309→                'DOM.describeNode',
   310→                { objectId: result.objectId, depth: -1, pierce: true },
   311→                session,
   312→              )) as { node: DomNode };
   313→              walk(node, frame.id);
   314→            }
   315→          }

   335→                session,

   341→          results.push({
   342→            origin: frame.securityOrigin ?? '',
   343→            session,

   347→        } catch (error) {

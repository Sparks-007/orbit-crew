    94→function toQuestionOption(value: unknown): QuestionOption | null {
    95→    if (typeof value === 'string') return value.trim() ? { label: value } : null
    96→    if (!value || typeof value !== 'object') return null
    97→    const { label, description } = value as { label?: unknown; description?: unknown }
    98→    if (typeof label !== 'string' || !label.trim()) return null
    99→    return typeof description === 'string' && description.trim() ? { label, description } : { label }
   100→}

   102→function toQuestionItem(question: unknown, options: unknown, multiple: unknown, header?: unknown): QuestionItem {
   103→    const choices = Array.isArray(options) ? options.map(toQuestionOption).filter(option => option !== null) : []
   104→    return {
   105→        question: String(question ?? 'The coworker has a question'),
   106→        ...(typeof header === 'string' && header.trim() ? { header } : {}),
   107→        options: choices,
   108→        multiple: multiple === true && choices.length > 0,
   109→    }
   110→}

   112→export function pendingQuestions(input: Record<string, unknown>): QuestionItem[] {
   113→    const questions = coerceJsonArray(input.questions)
   114→    if (Array.isArray(questions)) {
   115→        return questions.map(value => {

   117→            const options = Array.isArray(item.options) ? item.options : item.suggestions

   130→function questionsCarryOptionObjects(input: Record<string, unknown>): boolean {
   131→    const questions = coerceJsonArray(input.questions)
   132→    if (!Array.isArray(questions)) return false
   133→    return questions.some(value => {
   134→        const item = (value ?? {}) as { options?: unknown; suggestions?: unknown }
   135→        const options = Array.isArray(item.options) ? item.options : item.suggestions
   136→        return Array.isArray(options) && options.some(option => option !== null && typeof option === 'object')
   137→    })
   138→}

   140→export function withHitlVersion<T>(toolName: string | undefined, input: T): T {
   141→    if (!toolName || hitlKindOf(toolName) !== 'question') return input
   142→    if (!input || typeof input !== 'object' || Array.isArray(input)) return input
   143→    const record = input as Record<string, unknown>
   144→    if (hitlVersionOf(record) >= QUESTION_OPTIONS_VERSION || !questionsCarryOptionObjects(record)) return input
   145→    return { ...record, v: QUESTION_OPTIONS_VERSION } as T
   146→}

   150→// Each block names its own files so the agent knows which answer they belong to, while the paths
   151→// travel once in the trailing marker, which is what renders them as tiles in the bubble.
   152→export function serializeAnswers(answers: QuestionAnswer[]): string {
   153→    const blocks = answers.map(a => {
   154→        const values = a.answers.map(value => value.trim()).filter(Boolean)
   155→        const files = a.attachments ?? []
   156→        if (values.length === 0 && files.length === 0) return `**${a.question}**\n[skipped]`
   157→        const lines = values.length > 1 ? values.map(value => `- ${value}`) : values
   158→        if (files.length > 0) lines.push(`Attached: ${files.map(attachmentDisplayName).join(', ')}`)
   159→        return `**${a.question}**\n${lines.join('\n')}`
   160→    })
   161→    const text = `Here are my answers:\n\n${blocks.join('\n\n')}`
   162→    const paths = answers.flatMap(a => a.attachments ?? [])
   163→    return paths.length > 0 ? appendAttachmentsRef(text, buildAttachmentsRef(paths)) : text
   164→}

   274→type HitlPart = { type: string; toolCallId?: string; toolName?: string; input?: unknown; rawInput?: unknown; output?: unknown }

   276→export type PendingHitl = { kind: HitlKind; toolCallId: string; input: Record<string, unknown>; output: unknown }
   277→
   278→// The question, approval, secret... the agent stopped on, when the transcript ends on one: the last
   279→// HITL call of the closing assistant message. Shared by the web and the app chat. The caller still
   280→// rules out a live stream and a session already idle, whose block was closed.
   281→export function findPendingHitl(messages: readonly { role: string; parts: readonly unknown[] }[], dismissedCallId: string | null = null): PendingHitl | null {
   282→    const last = messages[messages.length - 1]
   283→    if (!last || last.role !== 'assistant') return null
   284→    for (let index = last.parts.length - 1; index >= 0; index--) {
   285→        const part = last.parts[index] as HitlPart
   286→        const toolName = part.toolName ?? (part.type.startsWith('tool-') ? part.type.slice('tool-'.length) : '')
   287→        const kind = hitlKindOf(toolName)
   288→        if (!kind || !part.toolCallId) continue
   289→        // A call rejected by input validation keeps its payload in `rawInput`, the AI SDK moving it
   290→        // out of `input`. Recover it so the form still holds the real questions, and skip the call
   291→        // entirely when nothing usable is left.
   292→        const input = (part.input ?? part.rawInput ?? {}) as Record<string, unknown>
   293→        if (!hitlInputIsUsable(kind, input)) return null
   294→        if (part.toolCallId === dismissedCallId) return null
   295→        // A spend the hub could not price never reached the user: the coworker read the error instead.
   296→        if (kind === 'spend' && !readSpendQuote(part.output)) return null
   297→        return { kind, toolCallId: part.toolCallId, input, output: part.output }
   298→    }
   299→    return null
   300→}

   302→// How a HITL input grows without stranding a client that shipped earlier, the iOS app above all,
   303→// which only updates when its users do: a field is never removed nor given another meaning, and a
   304→// shape a client could not read the old way takes a new version in `v` (absent means 1). A client
   305→// draws the versions it knows and, past them, shows what is asked as text to be answered in its
   306→// composer.
   307→export function hitlVersionOf(input: Record<string, unknown>): number {
   308→    const version = input.v
   309→    return typeof version === 'number' && Number.isInteger(version) && version > 0 ? version : 1
   310→}

   312→export type UnknownPendingCall = { toolCallId: string; toolName: string; input: Record<string, unknown> }

   314→// The tool call the closing assistant message stopped on, still without a result, when it is no HITL
   315→// this client knows: one an engine newer than the client pauses on. A run that died mid-call ends the
   316→// same way, so the caller only asks once the session says it waits on a human.
   317→export function findUnknownPendingCall(messages: readonly { role: string; parts: readonly unknown[] }[]): UnknownPendingCall | null {
   318→    const last = messages[messages.length - 1]
   319→    if (!last || last.role !== 'assistant') return null
   320→    for (let index = last.parts.length - 1; index >= 0; index--) {
   321→        const part = last.parts[index] as HitlPart & { state?: string }
   322→        const isCall = part.type === 'dynamic-tool' || part.type.startsWith('tool-')
   323→        if (!isCall) continue
   324→        const toolName = stripBridgePrefix(part.toolName ?? part.type.slice('tool-'.length))
   325→        if (!part.toolCallId || hitlKindOf(toolName) || part.output !== undefined || part.state === 'output-available' || part.state === 'output-error') return null
   326→        return { toolCallId: part.toolCallId, toolName, input: (part.input ?? part.rawInput ?? {}) as Record<string, unknown> }
   327→    }
   328→    return null
   329→}

   331→export type PendingAlert = { toolCallId: string | null; message: string }

   333→const ALERT_FALLBACK_MESSAGE = 'The coworker sent you a notification.'
   334→
   335→// The notification the session is waiting on: the last `notify_user` raised with kind `info`. It is
   336→// not necessarily the last tool call, since it does not stop the run. The status in the database is
   337→// what decides whether one is pending, so a transcript that no longer holds the call (history
   338→// trimmed, notification raised in an earlier run) still gets a block to acknowledge.
   339→export function findPendingAlert(messages: readonly { role: string; parts: readonly unknown[] }[]): PendingAlert {
   340→    for (let i = messages.length - 1; i >= 0; i--) {
   341→        const message = messages[i]
   342→        if (message.role !== 'assistant') continue
   343→        for (let j = message.parts.length - 1; j >= 0; j--) {
   344→            const part = message.parts[j] as { type: string; toolCallId?: string; toolName?: string; input?: unknown }
   345→            if (typeof part?.type !== 'string' || toolNameOf(part) !== 'notify_user') continue
   346→            const input = (part.input ?? {}) as Record<string, unknown>
   347→            // `kind` defaults to `info` in the tool schema, so a call that left it out is one of these.
   348→            if ((input.kind ?? 'info') !== 'info') continue
   349→            return { toolCallId: part.toolCallId ?? null, message: typeof input.message === 'string' && input.message.trim() ? input.message : ALERT_FALLBACK_MESSAGE }
   350→        }
   351→    }
   352→    return { toolCallId: null, message: ALERT_FALLBACK_MESSAGE }
   353→}

   369→const SERVICE_RESULT = /^App setup result:\n([\s\S]*)\n\nPlease continue\.$/
   370→const SERVICE_SUFFIXES: [string, ServiceSetupOutcome][] = [
   371→    [': connected', 'connected'],
   372→    [': connection reset, not reconnected', 'reset'],
   373→    [': not found in the catalog', 'not-found'],
   374→    [': skipped', 'skipped'],
   375→]
   376→const INSTEAD_OF = /^(.*) \(instead of (.*)\)$/
   377→
   378→export type ServiceSetupEntry = { name: string; outcome: ServiceSetupOutcome; replaces?: string }
   379→
   380→function parseServiceLine(line: string): ServiceSetupEntry | null {
   381→    if (!line.startsWith('- ')) return null
   382→    const body = line.slice(2)
   383→    const match = SERVICE_SUFFIXES.find(([suffix]) => body.endsWith(suffix))
   384→    if (!match) return null
   385→    const [suffix, outcome] = match
   386→    const label = body.slice(0, -suffix.length)
   387→    const replaced = outcome === 'not-found' ? null : INSTEAD_OF.exec(label)
   388→    return replaced ? { name: replaced[1], outcome, replaces: replaced[2] } : { name: label, outcome }
   389→}
   390→
   391→// A `not-found` entry carries the slug the agent asked for, since the catalog had no name for it.
   392→export function parseServiceSetupResult(text: string): ServiceSetupEntry[] | null {
   393→    const match = SERVICE_RESULT.exec(text.trim())
   394→    if (!match) return null
   395→    if (match[1] === '- (no apps suggested)') return []
   396→    const entries = match[1].split('\n').map(parseServiceLine)
   397→    return entries.every(entry => entry !== null) ? (entries as ServiceSetupEntry[]) : null
   398→}

   400→const SECRET_RESULT = /^Credential setup result:\n([\s\S]*)\n\nReference a provided variable as/
   401→// Transcripts written before the copy said "coworker" still carry "an agent variable".
   402→const SECRET_LINE = /^- (\S+): (provided \(stored as (?:an agent|a coworker) variable\)|skipped)$/
   403→
   404→export function parseSecretResult(text: string): { key: string; provided: boolean }[] | null {
   405→    const match = SECRET_RESULT.exec(text.trim())
   406→    if (!match) return null
   407→    const entries = match[1].split('\n').map(line => SECRET_LINE.exec(line))
   408→    if (!entries.every(entry => entry !== null)) return null
   409→    return (entries as RegExpExecArray[]).map(entry => ({ key: entry[1], provided: entry[2] !== 'skipped' }))
   410→}

   419→const APPROVAL_RESULT = /^Here are my approval decisions:\n\n([\s\S]*)\n\nPerform only the approved actions\./
   420→const APPROVAL_BLOCK_START = /\n\n(?=\d+\. \*\*(?:APPROVED|DECLINED)\*\* — )/
   421→const APPROVAL_BLOCK = /^\d+\. \*\*(APPROVED|DECLINED)\*\* — ([^\n]*)(?:\nReason: ([\s\S]*))?$/
   422→
   423→export function parseApprovalDecisions(text: string): Omit<ApprovalDecision, 'detail'>[] | null {
   424→    const match = APPROVAL_RESULT.exec(text.trim())
   425→    if (!match) return null
   426→    const blocks = match[1].split(APPROVAL_BLOCK_START).map(block => APPROVAL_BLOCK.exec(block))
   427→    if (!blocks.every(block => block !== null)) return null
   428→    return (blocks as RegExpExecArray[]).map(([, status, action, note]): Omit<ApprovalDecision, 'detail'> => (note === undefined ? { action, decision: status === 'APPROVED' ? 'approved' : 'declined' } : { action, decision: 'declined', note }))
   429→}

   431→export function pendingSpend(input: Record<string, unknown>): SpendRequest {

   436→const SPEND_ALWAYS_LINE = 'Always allowed: paid data tools no longer ask for this coworker.'
   437→const SPEND_APPROVED_NEXT = 'Go on with the paid calls you planned, and stay close to the estimate.'
   438→const SPEND_DECLINED_NEXT = 'Do not call any paid data tool for this. Go on with the free tools if they can do part of it, or ask me what to change.'

   440→export function serializeSpendDecision({ purpose, decision, always }: SpendDecision): string {
   441→    const head = `Spend decision: **${decision === 'approved' ? 'APPROVED' : 'DECLINED'}** — ${purpose.replace(/\s*\n\s*/g, ' ')}`
   442→    if (decision === 'declined') return `${head}\n\n${SPEND_DECLINED_NEXT}`
   443→    return `${head}${always ? `\n${SPEND_ALWAYS_LINE}` : ''}\n\n${SPEND_APPROVED_NEXT}`
   444→}

   446→// A decline written between 14 and 22 October 2026 could carry a reason, up to the blank line before
   447→// the closing instruction: still read as a decline, the reason left out.
   448→const SPEND_RESULT = /^Spend decision: \*\*(APPROVED|DECLINED)\*\* — ([^\n]*)(?:\nReason: [\s\S]*?)?(\nAlways allowed: paid data tools no longer ask for this coworker\.)?\n\n(?:Go on with the paid calls|Do not call any paid data tool)/

   450→export function parseSpendDecision(text: string): SpendDecision | null {
   451→    const match = SPEND_RESULT.exec(text.trim())
   452→    if (!match) return null
   453→    const [, status, purpose, always] = match
   454→    if (status === 'DECLINED') return { purpose, decision: 'declined' }
   455→    return always ? { purpose, decision: 'approved', always: true } : { purpose, decision: 'approved' }
   456→}

   458→const ANSWERS_HEADER = 'Here are my answers:\n\n'
   459→const QUESTION_LINE = /^\*\*(.+)\*\*$/
   460→const ATTACHED_LINE = /^Attached: (.+)$/
   461→
   462→// A block starts at a bold line after a blank one. An answer typed with a line of its own in bold
   463→// would split there, which only costs that one answer its layout.
   464→export function parseAnswers(text: string): QuestionAnswer[] | null {
   465→    const withFiles = parseAttachments(text)
   466→    const body = (withFiles?.text ?? text).trim()
   467→    if (!body.startsWith(ANSWERS_HEADER)) return null
   468→    const paths = withFiles?.paths ?? []
   469→    const blocks = body.slice(ANSWERS_HEADER.length).split(/\n\n(?=\*\*.+\*\*\n)/)
   470→    const answers: QuestionAnswer[] = []
   471→    for (const block of blocks) {
   472→        const [heading, ...lines] = block.split('\n')
   473→        const question = QUESTION_LINE.exec(heading)
   474→        if (!question || lines.length === 0) return null
   475→        if (lines.length === 1 && lines[0] === '[skipped]') {
   476→            answers.push({ question: question[1], answers: [] })
   477→            continue
   478→        }
   479→        const attached = ATTACHED_LINE.exec(lines.at(-1) ?? '')
   480→        const values = attached ? lines.slice(0, -1) : lines
   481→        const listed = values.length > 1 && values.every(line => line.startsWith('- '))
   482→        const answer: QuestionAnswer = { question: question[1], answers: listed ? values.map(line => line.slice(2)) : values.length > 0 ? [values.join('\n')] : [] }
   483→        if (attached) {
   484→            const names = attached[1].split(', ')
   485→            answer.attachments = paths.filter(path => names.includes(attachmentDisplayName(path)))
   486→        }
   487→        answers.push(answer)
   488→    }
   489→    return answers
   490→}

   492→export type HitlReply =
   493→    | { kind: 'service'; entries: ServiceSetupEntry[] }
   494→    | { kind: 'question'; answers: QuestionAnswer[] }
   495→    | { kind: 'secret'; entries: { key: string; provided: boolean }[] }
   496→    | { kind: 'mcp'; name: string }
   497→    | { kind: 'approval'; decisions: Omit<ApprovalDecision, 'detail'>[] }
   498→    | { kind: 'takeover'; url: string | null }
   499→    | ({ kind: 'spend' } & SpendDecision)
   500→    | { kind: 'cancelled'; of: keyof typeof HITL_CANCELLED }
   501→
   502→export function parseHitlReply(text: string): HitlReply | null {
   503→    const trimmed = text.trim()
   504→    const cancelled = (Object.keys(HITL_CANCELLED) as (keyof typeof HITL_CANCELLED)[]).find(kind => HITL_CANCELLED[kind] === trimmed)
   505→    if (cancelled) return { kind: 'cancelled', of: cancelled }
   506→    const services = parseServiceSetupResult(trimmed)
   507→    if (services) return { kind: 'service', entries: services }
   508→    const answers = parseAnswers(trimmed)
   509→    if (answers) return { kind: 'question', answers }
   510→    const secrets = parseSecretResult(trimmed)
   511→    if (secrets) return { kind: 'secret', entries: secrets }
   512→    const mcp = parseMcpSignInResult(trimmed)
   513→    if (mcp) return { kind: 'mcp', name: mcp.name }
   514→    const decisions = parseApprovalDecisions(trimmed)
   515→    if (decisions) return { kind: 'approval', decisions }
   516→    const takeover = parseTakeoverResult(trimmed)
   517→    if (takeover) return { kind: 'takeover', url: takeover.url }
   518→    const spend = parseSpendDecision(trimmed)
   519→    if (spend) return { kind: 'spend', ...spend }
   520→    return null
   521→}

    52→function describeFailure(error: unknown, fallback: string) {
    53→  return {
    54→    ok: false as const,
    55→    status: error instanceof ApiError ? error.status : undefined,
    56→    message: error instanceof Error ? error.message : fallback,
    57→  };
    58→}

    61→  const [state, setState] = useState<State>();
    62→  const [workspace, setWorkspace] = useState<WorkspaceState>();
    63→  const [selectedDot, setSelectedDot] = useState('');
    64→  const [selectedThread, setSelectedThread] = useState<string>();
    65→  const [view, rawSetView] = useState<'chat' | 'tasks' | 'memories' | 'space'>(
    66→    'chat',
    67→  );
    68→  const dirtyPage = useRef(false);
    69→  const [spaceId, setSpaceId] = useState('');
    70→  const [pageId, setPageId] = useState<string>();
    71→  const setDirtyPage = useCallback((value: boolean) => {
    72→    dirtyPage.current = value;
    73→  }, []);
    74→  const setView = (next: typeof view) => {

    78→    if (next !== 'space')
    79→      history.replaceState(null, '', location.pathname + location.search);
    80→    rawSetView(next);
    81→  };
    82→  useEffect(() => {
    83→    let acceptedHash = location.hash;
    84→    const navigate = () => {
    85→      const match = location.hash.match(
    86→        /^#\/spaces\/([^/]+)(?:\/pages\/([^/]+))?$/,
    87→      );
    88→      if (!match) return;
    89→      if (dirtyPage.current && location.hash === acceptedHash) return;
    90→      if (
    91→        dirtyPage.current &&
    92→        !window.confirm('Leave your unsaved page draft?')
    93→      ) {
    94→        history.replaceState(null, '', acceptedHash || location.pathname);
    95→        return;
    96→      }
    97→      acceptedHash = location.hash;
    98→      dirtyPage.current = false;
    99→      setSpaceId(match[1]);
   100→      setPageId(match[2]);
   101→      rawSetView('space');
   102→      setMobile(false);
   103→    };
   104→    navigate();
   105→    window.addEventListener('hashchange', navigate);
   106→    return () => window.removeEventListener('hashchange', navigate);
   107→  }, []);
   108→  const openPage = (space: string, page?: string) => {
   109→    openPageLink(`#/spaces/${space}${page ? `/pages/${page}` : ''}`);
   110→  };
   111→
   112→  const [notices, setNotices] = useState<Notices>({
   113→    refresh: '',
   114→    capture: '',
   115→    action: '',
   116→  });
   117→  const error = visibleNotice(notices);
   118→  const setError = (action: string) =>
   119→    setNotices((current) =>
   120→      current.action === action ? current : { ...current, action },
   121→    );
   122→  const [auth, setAuth] = useState('');
   123→  const [needsAuth, setNeedsAuth] = useState(false);
   124→  const [dialog, setDialog] = useState<Dialog>();
   125→  const [mobile, setMobile] = useState(false);
   126→  const [navCollapsed, setNavCollapsed] = useState(false);
   127→  const [pane, setPane] = useState(false);
   128→  const [capture, setCapture] = useState<Result>();
   129→  const [prompt, setPrompt] = useState('');
   130→  const [pendingPrompt, setPendingPrompt] = useState<string>();
   131→  const [busy, setBusy] = useState(false);
   132→  const [search, setSearch] = useState('');
   133→  const [taskDetail, setTaskDetail] = useState<Detail>();
   134→  const refresh = useCallback(async () => {
   135→    try {
   136→      const [s, w] = await Promise.all([
   137→        api<State>('/state'),
   138→        api<WorkspaceState>('/workspace'),
   139→      ]);
   140→      setState(s);
   141→      setWorkspace(w);
   142→      setNeedsAuth(false);
   143→      setSelectedDot((previous) => previous || w.dots[0]?.id || '');
   144→      setNotices((current) => applyRefreshResult(current, { ok: true }));
   145→    } catch (e) {
   146→      if (e instanceof ApiError && e.status === 401) setNeedsAuth(true);
   147→      setNotices((current) =>
   148→        applyRefreshResult(
   149→          current,
   150→          describeFailure(e, 'Could not connect to the server.'),
   151→        ),
   152→      );
   153→    }
   154→  }, []);
   155→  useEffect(() => {
   156→    void refresh();
   157→    const timer = setInterval(() => void refresh(), 3000);
   158→    return () => clearInterval(timer);
   159→  }, [refresh]);
   160→  useEffect(() => {
   161→    setCapture(undefined);
   162→    if (!selectedThread) return;
   163→    let active = true;
   164→    const load = () =>
   165→      void api<Result | null>(`/conversations/${selectedThread}/capture`)
   166→        .then((result) => {
   167→          if (!active) return;
   168→          setCapture(result ?? undefined);
   169→          setNotices((current) => applyCaptureResult(current, { ok: true }));
   170→        })
   171→        .catch((e) => {
   172→          if (!active) return;
   173→          setNotices((current) =>
   174→            applyCaptureResult(
   175→              current,
   176→              describeFailure(e, 'Could not connect to the server.'),
   177→            ),
   178→          );
   179→        });
   180→    load();
   181→    const timer = setInterval(load, 3000);
   182→    return () => {
   183→      active = false;
   184→      clearInterval(timer);
   185→    };
   186→  }, [selectedThread]);
   187→  const mutate = async (path: string, method: string, body?: unknown) => {
   188→    setError('');
   189→    try {
   190→      await api(path, method, body);
   191→      await refresh();
   192→      if (taskDetail)
   193→        setTaskDetail(await api<Detail>(`/tasks/${taskDetail.task.id}`));
   194→      return true;
   195→    } catch (e) {
   196→      setError(e instanceof Error ? e.message : 'Could not save.');
   197→      return false;
   198→    }
   199→  };
   200→  const dot =
   201→    workspace?.dots.find((item) => item.id === selectedDot) ??
   202→    workspace?.dots[0];
   203→  const thread = workspace?.conversations.find(
   204→    (item) => item.id === selectedThread && item.dotId === dot?.id,
   205→  );
   206→  const configured = !!workspace && workspace.setup.missing.length === 0;
   207→  const setupStep = workspace
   208→    ? dialog?.type === 'settings'
   209→      ? 'settings'
   210→      : configured
   211→        ? 'ready'
   212→        : 'setup_required'
   213→    : undefined;
   214→  useEffect(() => {
   215→    if (!setupStep) return;
   216→    return trackSetupStep(setupStep);
   217→  }, [setupStep]);
   218→  const chooseDot = (next: Dot) => {
   219→    setSelectedDot(next.id);
   220→    setSelectedThread(
   221→      workspace?.conversations.find((item) => item.dotId === next.id)?.id,
   222→    );
   223→    setView('chat');
   224→    setMobile(false);
   225→    setPendingPrompt(undefined);
   226→  };

   228→    if (!dot || !configured || busy) return;
   229→    setBusy(true);
   230→    setError('');
   231→    try {
   232→      const next = await api<Conversation>('/conversations', 'POST', {
   233→        dotId: dot.id,
   234→        title: text?.slice(0, 80) || 'A new thought',
   235→      });
   236→      await refresh();
   237→      setSelectedThread(next.id);
   238→      setPendingPrompt(text);
   239→      setPrompt('');
   240→      setView('chat');
   241→      setMobile(false);
   242→    } catch (e) {
   243→      setError(
   244→        e instanceof Error ? e.message : 'Could not create the conversation.',
   245→      );
   246→    } finally {
   247→      setBusy(false);
   248→    }
   249→  };
   250→  if (needsAuth)
   251→    return (
   252→      <main className="unlock">
   253→        <Mascot />
   254→        <h1>Your own little corner.</h1>
   255→        <p>
   256→          Enter the owner access token configured on this template’s server.
   257→        </p>
   258→        <form
   259→          onSubmit={async (e) => {
   260→            e.preventDefault();
   261→            setToken(auth);
   262→            try {
   263→              await api('/state');
   264→              setError('');
   265→              await refresh();
   266→            } catch (err) {
   267→              setError(
   268→                err instanceof Error
   269→                  ? err.message
   270→                  : 'Access token was not accepted.',
   271→              );
   272→            }
   273→          }}
   274→        >
   275→          <input
   276→            type="password"
   277→            aria-label="Owner access token"
   278→            autoComplete="current-password"
   279→            value={auth}
   280→            onChange={(e) => setAuth(e.target.value)}
   281→            required
   282→          />
   283→          <button className="primary">Unlock OpenDots</button>
   284→        </form>
   285→        {error && (
   286→          <p className="chat-error" role="alert">
   287→            {error}
   288→          </p>
   289→        )}
   290→        <p className="muted">The token stays in this tab’s session storage.</p>
   291→      </main>
   292→    );
   293→  if (!state || !workspace || !dot)
   294→    return (
   295→      <main className="unlock">
   296→        <Mascot state="working" />
   297→        <h1>Finding your dots…</h1>
   298→        {error && (
   299→          <>
   300→            <p className="chat-error">{error}</p>
   301→            <button onClick={() => void refresh()}>Retry</button>
   302→          </>
   303→        )}
   304→      </main>
   305→    );
   306→  const content = (
   307→    <div className={`app template-app ${navCollapsed ? 'nav-collapsed' : ''}`}>
   308→      <nav className="icon-rail" aria-label="Workspace navigation">
   309→        <button
   310→          className="rail-brand"
   311→          aria-label="OpenDots home"
   312→          onClick={() => {
   313→            setView('chat');
   314→            setSelectedThread(undefined);
   315→          }}
   316→        >
   317→          o<span>·</span>
   318→        </button>
   319→        <button
   320→          aria-label={navCollapsed ? 'Expand sidebar' : 'Collapse sidebar'}
   321→          onClick={() => setNavCollapsed(!navCollapsed)}
   322→        >
   323→          <PanelLeft size={18} />
   324→        </button>
   325→        <button
   326→          aria-label="New chat"
   327→          disabled={!configured}
   328→          onClick={() => void newConversation()}
   329→        >
   330→          <Plus size={19} />
   331→        </button>
   332→        <button
   333→          aria-label="Open Spaces"
   334→          onClick={() => {
   335→            if (workspace.spaces[0]) openPage(workspace.spaces[0].id);
   336→          }}
   337→        >
   338→          <Folder size={18} />
   339→        </button>
   340→        <button aria-label="Open activity" onClick={() => setView('tasks')}>
   341→          <Clock3 size={18} />
   342→        </button>
   343→        <button
   344→          className="rail-settings"
   345→          aria-label="Open settings"
   346→          onClick={() => setDialog({ type: 'settings' })}
   347→        >
   348→          <Settings2 size={18} />
   349→        </button>
   350→      </nav>
   351→      <button
   352→        className="mobile-menu icon-button"
   353→        aria-label="Open navigation"
   354→        aria-expanded={mobile}
   355→        aria-controls="workspace-sidebar"
   356→        onClick={() => setMobile(true)}
   357→      >
   358→        <Menu size={21} />
   359→      </button>
   360→      {mobile && (
   361→        <button
   362→          className="nav-scrim"
   363→          aria-label="Close navigation"
   364→          onClick={() => setMobile(false)}
   365→        />
   366→      )}
   367→      <aside
   368→        id="workspace-sidebar"
   369→        className={`sidebar ${mobile ? 'open' : ''}`}
   370→      >
   371→        <button
   372→          className="wordmark"
   373→          onClick={() => {
   374→            setView('chat');
   375→            setSelectedThread(undefined);
   376→          }}
   377→        >
   378→          <span className="dotted-logo">
   379→            <i />
   380→            <i />
   381→            <i />
   382→            <i />
   383→          </span>
   384→          OpenDots<span className="wordmark-dot">•</span>
   385→        </button>
   386→        <button
   387→          className="new-chat nav-item"
   388→          disabled={!configured}
   389→          onClick={() => void newConversation()}
   390→        >
   391→          <Plus size={17} />
   392→          <span>New chat</span>
   393→        </button>
   394→        <div className="spaces-heading nav-label">
   395→          DOTS
   396→          <button
   397→            className="icon-button"
   398→            aria-label="Create Dot"
   399→            onClick={() =>
   400→              setDialog({ type: 'dot', spaceId: workspace.spaces[0].id })
   401→            }
   402→          >
   403→            <Plus size={14} />
   404→          </button>
   405→        </div>
   406→        <nav className="dots-nav" aria-label="Dots">
   407→          {workspace.dots.map((item) => (
   408→            <div className="dot-nav-row" key={item.id}>
   409→              <button
   410→                className={`dot-nav ${dot.id === item.id && view === 'chat' ? 'active' : ''}`}
   411→                aria-current={
   412→                  dot.id === item.id && view === 'chat' ? 'page' : undefined
   413→                }
   414→                onClick={() => chooseDot(item)}
   415→              >
   416→                <Mascot identity={item.id} name={item.name} small decorative />
   417→                <span>{item.name}</span>
   418→              </button>
   419→              <button
   420→                className="icon-button dot-settings"
   421→                aria-label={`Edit ${item.name} settings`}
   422→                onClick={() =>
   423→                  setDialog({ type: 'dot', dot: item, spaceId: item.spaceId })
   424→                }
   425→              >
   426→                <MoreHorizontal size={15} />
   427→              </button>
   428→            </div>
   429→          ))}
   430→        </nav>
   431→        <div className="spaces-heading nav-label">
   432→          SPACES
   433→          <button
   434→            className="icon-button"
   435→            aria-label="Create Space"
   436→            onClick={() => setDialog({ type: 'space' })}
   437→          >
   438→            <Plus size={14} />
   439→          </button>
   440→        </div>
   441→        <nav className="spaces-nav" aria-label="Spaces">
   442→          {workspace.spaces.map((space) => (
   443→            <SpaceNav
   444→              key={space.id}
   445→              space={space}
   446→              active={view === 'space' && spaceId === space.id}
   447→              pageId={pageId}
   448→              onOpen={(id) => openPage(space.id, id)}
   449→            />
   450→          ))}
   451→        </nav>
   452→        {configured ? (
   453→          <ThreadList
   454→            dotId={dot.id}
   455→            dots={workspace.dots}
   456→            local={workspace.conversations}
   457→            selected={view === 'chat' ? selectedThread : undefined}
   458→            onSelect={(id) => {
   459→              const conversation = workspace.conversations.find(
   460→                (item) => item.id === id,
   461→              );
   462→              if (conversation) setSelectedDot(conversation.dotId);
   463→              setSelectedThread(id);
   464→              setView('chat');
   465→              setMobile(false);
   466→            }}
   467→            onNew={() => void newConversation()}
   468→          />
   469→        ) : (
   470→          <div className="sidebar-empty">
   471→            Set up text chat to begin a persistent conversation.
   472→          </div>
   473→        )}
   474→        <div className="sidebar-bottom">
   475→          <button
   476→            className={`nav-item ${view === 'tasks' ? 'active' : ''}`}
   477→            onClick={() => {
   478→              setView('tasks');
   479→              setMobile(false);
   480→            }}
   481→          >
   482→            <Clock3 size={17} />
   483→            <span>Scheduled & activity</span>
   484→            {state.tasks.length > 0 && <small>{state.tasks.length}</small>}
   485→          </button>
   486→          <button
   487→            className={`nav-item ${view === 'memories' ? 'active' : ''}`}
   488→            onClick={() => {
   489→              setView('memories');
   490→              setMobile(false);
   491→            }}
   492→          >
   493→            <BookOpen size={17} />
   494→            <span>Memories</span>
   495→            {state.memories.length > 0 && (
   496→              <small>{state.memories.length}</small>
   497→            )}
   498→          </button>
   499→          <button
   500→            className="nav-item"
   501→            onClick={() => setDialog({ type: 'settings' })}
   502→          >
   503→            <Settings2 size={17} />
   504→            <span>Settings & setup</span>
   505→          </button>
   506→          <a
   507→            className="nav-item"
   508→            href="https://github.com/CopilotKit/OpenDots"
   509→            target="_blank"
   510→            rel="noreferrer"
   511→          >
   512→            <Code2 size={17} />
   513→            <span>Make it your own</span>
   514→            <ArrowUpRight size={13} />
   515→          </a>
   516→          <div className="version">
   517→            OPEN SOURCE TEMPLATE <span>v0.1</span>
   518→          </div>
   519→        </div>
   520→      </aside>
   521→      <div className="workspace">
   522→        <header className="topbar">
   523→          <button
   524→            className="desktop-nav-toggle document-icon"
   525→            aria-label={navCollapsed ? 'Show navigation' : 'Hide navigation'}
   526→            onClick={() => setNavCollapsed(!navCollapsed)}
   527→          >
   528→            <PanelLeft size={18} />
   529→          </button>
   530→          <div className="breadcrumbs">
   531→            <span>
   532→              {view === 'space'
   533→                ? workspace.spaces.find((space) => space.id === spaceId)?.name
   534→                : 'Dots'}
   535→            </span>
   536→            <span>/</span>
   537→            <strong>
   538→              {view === 'chat'
   539→                ? dot.name
   540→                : view === 'tasks'
   541→                  ? 'Activity'
   542→                  : view === 'space'
   543→                    ? 'Pages'
   544→                    : 'Memories'}
   545→            </strong>
   546→          </div>
   547→          <div className="top-actions">
   548→            <span className="mode-badge">
   549→              {configured ? 'SELF-HOSTED' : 'SETUP REQUIRED'}
   550→            </span>
   551→            <button
   552→              className="pause-button"
   553→              aria-label={
   554→                state.settings.paused ? 'Resume all Dots' : 'Pause all Dots'
   555→              }
   556→              onClick={() =>
   557→                void mutate('/settings', 'PATCH', {
   558→                  paused: !state.settings.paused,
   559→                })
   560→              }
   561→            >
   562→              {state.settings.paused ? <Play size={14} /> : <Pause size={14} />}
   563→              <span>{state.settings.paused ? 'Resume' : 'Pause'}</span>
   564→            </button>
   565→            <button
   566→              className="icon-button"
   567→              aria-label={pane ? 'Hide computer' : 'Show computer'}
   568→              aria-expanded={pane}
   569→              onClick={() => setPane(!pane)}
   570→            >
   571→              <Monitor size={18} />
   572→            </button>
   573→          </div>
   574→        </header>
   575→        {error && (
   576→          <div className="error-banner" role="alert">
   577→            <span>{error}</span>
   578→            <button
   579→              className="icon-button"
   580→              aria-label="Dismiss error"
   581→              onClick={() => setNotices((current) => dismissNotice(current))}
   582→            >
   583→              <X size={16} />
   584→            </button>
   585→          </div>
   586→        )}
   587→        {state.settings.paused && (
   588→          <div className="notice">
   589→            All Dots are paused. Active compute stops and scheduled tasks wait.
   590→          </div>
   591→        )}
   592→        {view === 'space' ? (
   593→          <SpaceWorkspace
   594→            key={spaceId}
   595→            space={
   596→              workspace.spaces.find((s) => s.id === spaceId) ??
   597→              workspace.spaces[0]
   598→            }
   599→            pageId={pageId}
   600→            workspace={workspace}
   601→            paused={state.settings.paused}
   602→            onPage={(id) => openPage(spaceId, id)}
   603→            onSettings={() => setDialog({ type: 'settings' })}
   604→            onCreateDot={() => setDialog({ type: 'dot', spaceId })}

   606→            onRefresh={refresh}
   607→            onSchedule={(threadId) => setDialog({ type: 'schedule', threadId })}
   608→            onThread={(id) => {

   612→                setSelectedDot(target.dotId);

   616→          />
   617→        ) : view === 'chat' ? (
   618→          <div className={`chat-workspace ${pane ? 'split' : ''}`}>
   619→            <div className="chat-column">
   620→              {thread && configured ? (
   621→                <Chat
   622→                  key={thread.id}
   623→                  thread={thread}
   624→                  dot={dot}
   625→                  initialPrompt={pendingPrompt}
   626→                  onConsumed={() => setPendingPrompt(undefined)}
   627→                  voiceReady={workspace.setup.voice}
   628→                  calls={workspace.calls.filter(
   629→                    (call) => call.threadId === thread.id,
   630→                  )}
   631→                  paused={state.settings.paused}
   632→                  onSaved={refresh}

   787→                <button

   936→                  ? (workspace.dots.find((candidate) =>
   937→                      candidate.spaceIds.includes(spaceId),

   941→              dotState="idle"

   946→      </div>
   947→      {dialog && (
   948→        <WorkspaceDialog
   949→          dialog={dialog}
   950→          state={state}

   956→    </div>
   957→  );
   958→  return configured ? (
   959→    <CopilotKitProvider runtimeUrl="/api/copilotkit" headers={authHeaders()}>
   960→      {content}

   962→  ) : (

   964→  );

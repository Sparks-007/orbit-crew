   148→export function localFileRules(
   149→  platform: NodeJS.Platform,
   150→  home: string,
   151→  dataRoot: string,
   152→  systemDrive = 'C:',
   153→) {
   154→  const path = platform === 'win32' ? nodePath.win32 : nodePath.posix;
   155→  const key = (value: string) => (platform === 'win32' ? value.toLowerCase() : value);
   156→  const fold = (value: string) => (platform === 'linux' ? value : value.toLowerCase());
   157→  function inside(root: string, target: string) {
   158→    const rel = path.relative(fold(root), fold(target));
   159→    return rel === '' || (!rel.startsWith('..') && !path.isAbsolute(rel));
   160→  }
   161→  function privateName(name: string) {
   162→    const lower = name.toLowerCase();
   163→    return (
   164→      name.startsWith('.') ||
   165→      (platform === 'win32' && windowsHidden.has(lower)) ||
   166→      privateFolders.has(lower) ||
   167→      secretNames.some((pattern) => pattern.test(name))
   168→    );
   169→  }
   170→  function anchor(target: string) {
   171→    if (inside(home, target)) return home;
   172→    if (platform === 'darwin' && fold(target) !== '/volumes' && inside('/Volumes', target))
   173→      return path.join('/Volumes', target.split(path.sep)[2]!);
   174→    if (
   175→      platform === 'win32' &&
   176→      /^[a-z]:\\/i.test(target) &&
   177→      key(target.slice(0, 2)) !== key(systemDrive)
   178→    )
   179→      return target.slice(0, 3);
   180→    if (platform === 'linux')
   181→      for (const mounts of ['/media', '/run/media']) {
   182→        const [user, drive] = path.relative(mounts, target).split(path.sep);
   183→        if (inside(mounts, target) && user && drive) return path.join(mounts, user, drive);
   184→      }
   185→  }
   186→  return {
   187→    inside,
   188→    same: (left: string, right: string) => key(left) === key(right),
   189→    privateName,
   190→    anchor,
   191→    holdsData: (target: string) => inside(target, dataRoot),
   192→    root: (target: string) => {
   193→      const base = anchor(target);
   194→      return base !== undefined && inside(target, base);
   195→    },
   196→    privatePath(target: string) {
   197→      const base = anchor(target);
   198→      return (
   199→        !base ||
   200→        inside(dataRoot, target) ||
   201→        path.relative(fold(base), fold(target)).split(path.sep).filter(Boolean).some(privateName)
   202→      );
   203→    },
   204→    program: (name: string) => programs[platform]?.test(name) ?? false,
   205→    executable: (name: string, mode: number) =>
   206→      (programs[platform]?.test(name) ?? false) || (platform !== 'win32' && (mode & 0o111) !== 0),
   207→  };
   208→}

   290→  async function resolvedAsFarAsItExists(path: string) {
   291→    const missing: string[] = [];

   293→      const real = await realpath(current).catch(() => undefined);

   296→      missing.push(nodePath.basename(current));
   297→    }

   350→  async function contained(
   351→    real: string,
   352→    keep: string[],
   353→    allowRoot: boolean,
   354→    shown: string,
   355→    write: boolean,
   356→  ) {
   357→    const rule = await rules;
   358→    if (
   359→      rule.privatePath(real) ||
   360→      (!allowRoot && rule.root(real)) ||
   361→      (write && rule.holdsData(real))
   362→    )
   363→      throw outside(shown);
   364→    if (keep.some((entry) => rule.inside(entry, real) || (write && rule.inside(real, entry))))
   365→      throw kept(shown);
   366→    return real;
   367→  }

   369→  async function existing(
   370→    path: string,
   371→    keep: string[],
   372→    allowRoot: boolean,
   373→    write: boolean,
   374→    shown = path,
   375→  ) {
   376→    const lexical = nodePath.resolve(path);
   377→    const rule = await rules;
   378→    if (rule.privatePath(lexical)) throw outside(shown);
   379→    if (keep.some((entry) => rule.inside(entry, lexical))) throw kept(shown);
   380→    try {
   381→      return await contained(await realpath(path), keep, allowRoot, shown, write);
   382→    } catch (error) {
   383→      if (error instanceof IntegrationError) throw error;
   384→      if (denied(error)) throw new IntegrationError(permissionMessage(path), 403);
   385→      throw new IntegrationError(`${path} does not exist.`, 404);
   386→    }
   387→  }

   389→  async function entry(path: string, keep: string[]) {
   390→    const parent = await existing(

   393→      true,
   394→      false,
   395→      path,

   441→  async function place(source: string, target: string, keep: string[]) {
   442→    await unchanged(target, keep);
   443→    const info = await lstat(source);
   444→    const current = await maybeLstat(target);
   445→    if (
   446→      current &&
   447→      info.ino &&
   448→      current.ino === info.ino &&
   449→      current.dev === info.dev &&
   450→      process.platform !== 'linux' &&
   451→      source !== target &&
   452→      source.toLowerCase() === target.toLowerCase()
   453→    )
   454→      return rename(source, target);
   455→    if (current) throw taken(target);
   456→    const fail = (error: NodeJS.ErrnoException) => {
   457→      throw error.code === 'EEXIST' ? taken(target) : error;
   458→    };
   459→    if (info.isFile()) {
   460→      const linked = await link(source, target).then(
   461→        () => true,
   462→        (error: NodeJS.ErrnoException) => (error.code === 'EEXIST' ? fail(error) : false),
   463→      );
   464→      if (!linked) await copyFile(source, target, constants.COPYFILE_EXCL).catch(fail);
   465→      return rm(source).catch(async (error) => {
   466→        await rm(target, { force: true });
   467→        throw error;
   468→      });
   469→    }
   470→    if (info.isDirectory() && process.platform !== 'win32') {
   471→      await mkdir(target).catch(fail);
   472→      return rename(source, target).catch(async (error: NodeJS.ErrnoException) => {
   473→        if (error.code !== 'EXDEV') {
   474→          await rmdir(target).catch(() => undefined);
   475→          throw error;
   476→        }
   477→        try {
   478→          for (const name of await readdir(source))
   479→            await cp(nodePath.join(source, name), nodePath.join(target, name), {
   480→              recursive: true,
   481→              errorOnExist: true,
   482→              force: false,
   483→              verbatimSymlinks: true,
   484→              preserveTimestamps: true,
   485→            });
   486→        } catch (copy) {
   487→          await rm(target, { recursive: true, force: true });
   488→          throw copy;
   489→        }
   490→        await rm(source, { recursive: true });
   491→      });
   492→    }
   493→    if (info.isSymbolicLink()) {
   494→      const pointer = await readlink(source);
   495→      if (!nodePath.isAbsolute(pointer) && nodePath.dirname(source) !== nodePath.dirname(target))
   496→        throw new IntegrationError(
   497→          `${source} is a relative link, so moving it to another folder would change where it points.`,
   498→        );
   499→      await symlink(pointer, target).catch(fail);
   500→      return rm(source).catch(async (error) => {
   501→        await rm(target, { force: true });
   502→        throw error;
   503→      });
   504→    }
   505→    if (process.platform !== 'win32')
   506→      throw new IntegrationError(`${source} is not a file, folder or link.`);
   507→    await rename(source, target);
   508→  }

   510→  async function regularFile(path: string) {
   511→    const info = await maybeLstat(path);
   512→    if (info && !info.isFile()) throw new IntegrationError(`${path} is not a regular file.`);
   513→    if (info && (await rules).executable(path, info.mode))
   514→      throw new IntegrationError(`${path} is executable. Nod only changes plain files.`);
   515→    return info;
   516→  }

   518→  async function replace(
   519→    path: string,
   520→    data: string | Uint8Array,
   521→    previous: Stats | undefined,
   522→    keep: string[],
   523→    signal?: AbortSignal,
   524→  ) {
   525→    if (typeof data === 'string' && Buffer.byteLength(data) > writeBytes)
   526→      throw new IntegrationError('The file text must be at most 1 MB. Nothing was changed.');
   527→    if (previous) await trashReady();
   528→    const folder = nodePath.dirname(path);
   529→    const temporary = nodePath.join(folder, `.nod-${randomUUID()}.tmp`);
   530→    await writeFile(temporary, data, { flag: 'wx' });
   531→    if (!previous)
   532→      return place(temporary, path, keep).catch(async (error) => {
   533→        await rm(temporary, { force: true });
   534→        throw error;
   535→      });
   536→    let backupFolder = '';
   537→    let backup = '';
   538→    try {
   539→      backupFolder = await mkdtemp(nodePath.join(folder, '.nod-'));
   540→      backup = nodePath.join(backupFolder, nodePath.basename(path));
   541→      await chmod(temporary, previous.mode & 0o7777);
   542→      await link(path, backup).catch(() => copyFile(path, backup));
   543→      await unchanged(path, keep);
   544→      await toTrash(path, signal);
   545→    } catch (error) {
   546→      await rm(temporary, { force: true });
   547→      const lost =
   548→        Boolean(backup) && !(await maybeLstat(path)) && Boolean(await maybeLstat(backup));
   549→      const restored =
   550→        lost &&
   551→        (await place(backup, path, keep).then(
   552→          () => true,
   553→          () => false,
   554→        ));
   555→      if (backupFolder) await rm(backupFolder, { recursive: true, force: true });
   556→      if (lost)
   557→        throw new IntegrationError(
   558→          restored
   559→            ? `${path} is unchanged. The Trash answered too late, so a copy of it may also be in the Trash.`
   560→            : `${path} went to the Trash, but its new version wasn't saved. It can be restored from the Trash.`,
   561→          409,
   562→        );
   563→      throw error;
   564→    }
   565→    try {
   566→      await place(temporary, path, keep);
   567→    } catch {
   568→      await rm(temporary, { force: true });
   569→      const restored = await place(backup, path, keep).then(
   570→        () => true,
   571→        () => false,
   572→      );
   573→      await rm(backupFolder, { recursive: true, force: true });
   574→      throw new IntegrationError(
   575→        restored
   576→          ? `${path} was not replaced and is unchanged. A copy of it is in the Trash.`
   577→          : `${path} was not replaced. Its previous version is in the Trash.`,
   578→        409,
   579→      );
   580→    }
   581→    await rm(backupFolder, { recursive: true, force: true });
   582→    return 'moved to the Trash';
   583→  }

   585→  async function quarantine(path: string) {
   586→    if (process.platform === 'win32')
   587→      return writeFile(`${path}:Zone.Identifier`, '[ZoneTransfer]\r\nZoneId=3\r\n').catch(
   588→        () => undefined,
   589→      );
   590→    if (process.platform !== 'darwin') return;
   591→    const stamp = Math.floor(Date.now() / 1000).toString(16);
   592→    await run('/usr/bin/xattr', ['-w', 'com.apple.quarantine', `0081;${stamp};Nod;`, path]).catch(
   593→      () => undefined,
   594→    );
   595→  }

   597→  async function readFile(path: string, limit: number) {
   598→    const checked = await maybeLstat(path);
   599→    if (!checked?.isFile()) throw new IntegrationError(`${path} is not a file.`);
   600→    const file = await open(path, constants.O_RDONLY | noFollow | noBlock);
   601→    try {
   602→      const opened = await file.stat();
   603→      if (opened.dev !== checked.dev || opened.ino !== checked.ino)
   604→        throw new IntegrationError(`${path} changed while Nod was opening it.`);
   605→      const size = opened.size;
   606→      const length = Math.min(size, limit);
   607→      const { buffer, bytesRead } = await file.read(Buffer.alloc(length), 0, length, 0);
   608→      return { size, bytes: buffer.subarray(0, bytesRead) };
   609→    } finally {
   610→      await file.close();
   611→    }
   612→  }

   614→  async function each<T>(items: T[], task: (item: T) => Promise<unknown>, signal?: AbortSignal) {
   615→    const done: unknown[] = [];
   616→    const failed: Array<{ item: T; error: string }> = [];
   617→    for (const item of items) {
   618→      signal?.throwIfAborted();
   619→      try {
   620→        done.push(await task(item));
   621→      } catch (error) {
   622→        failed.push({ item, error: error instanceof Error ? error.message : String(error) });
   623→      }
   624→    }
   625→    return { done, failed };
   626→  }

   628→  type Input = {
   629→    path: string;
   630→    moves: Array<{ from: string; to: string }>;
   631→    paths: string[];
   632→    text: string;
   633→    overwrite?: boolean;
   634→    edits: Array<{ oldText: string; newText: string }>;
   635→    file: z.infer<typeof workspaceFileReference>;
   636→  };

   638→  function validate(action: Action, args: Record<string, unknown>) {
   639→    const parsed = schemas[action].safeParse(args);
   640→    if (!parsed.success)
   641→      throw new IntegrationError(
   642→        `Invalid ${action} arguments: ${z.prettifyError(parsed.error)}. Nothing was changed.`,
   643→      );
   644→    return parsed.data as Partial<Input>;
   645→  }

   647→  async function prepare(action: Action, args: Record<string, unknown>) {
   648→    await assertEnabled();
   649→    const input = validate(action, args);
   650→    const real = resolver(action, await keptOut());
   651→    if (input.moves)
   652→      return {
   653→        moves: await Promise.all(
   654→          input.moves.map(async ({ from, to }) => ({ from: await real(from), to: await real(to) })),
   655→        ),
   656→      };
   657→    if (input.paths) return { paths: await Promise.all(input.paths.map(real)) };
   658→    return { ...input, path: await real(input.path!) };
   659→  }

   661→  async function execute(
   662→    action: Action,
   663→    args: Record<string, unknown>,
   664→    context: { signal?: AbortSignal; read?: WorkspaceRead } = {},
   665→  ) {
   666→    await assertEnabled();
   667→    const { signal } = context;
   668→    const input = validate(action, args) as Input;
   669→    signal?.throwIfAborted();
   670→    const keep = await keptOut();
   671→    const rule = await rules;
   672→    const resolve = resolver(action, keep);
   673→    const bound = (resolved: string, requested: string) => {
   674→      if (!rule.same(resolved, requested))
   675→        throw new IntegrationError(
   676→          `${requested} now leads somewhere else than when this action was checked. Nothing was changed.`,
   677→          409,
   678→        );
   679→      return resolved;
   680→    };
   681→    const target = async (path: string) => bound(await resolve(path), path);
   682→    const destination = async (path: string, source?: string) => {
   683→      const resolved = await target(path);
   684→      if (
   685→        (rule.program(resolved) &&
   686→          !(
   687→            source &&
   688→            nodePath.extname(source).toLowerCase() === nodePath.extname(resolved).toLowerCase()
   689→          )) ||
   690→        (process.platform === 'win32' && /[. ]$/.test(resolved))
   691→      )
   692→        throw new IntegrationError(
   693→          `${path} would be a program or shortcut. Nod only creates plain files.`,
   694→        );
   695→      return resolved;
   696→    };
   697→    if (action === 'LOCAL_FILES_LIST') {
   698→      const path = await target(input.path);
   699→      const names = (await readdir(path, { withFileTypes: true })).filter(
   700→        (item) =>
   701→          !rule.privateName(item.name) &&
   702→          !keep.some((entry) => rule.inside(entry, nodePath.join(path, item.name))),
   703→      );
   704→      const entries = [];
   705→      for (let start = 0; start < names.length; start += statBatch) {
   706→        signal?.throwIfAborted();
   707→        entries.push(
   708→          ...(await Promise.all(
   709→            names.slice(start, start + statBatch).map(async (item) => {
   710→              const info = await maybeLstat(nodePath.join(path, item.name));
   711→              return {
   712→                name: item.name,
   713→                kind: (info ?? item).isDirectory()
   714→                  ? 'folder'
   715→                  : (info ?? item).isSymbolicLink()
   716→                    ? 'link'
   717→                    : 'file',
   718→                size: info?.size,
   719→                modified: info?.mtime.toISOString(),
   720→              };
   721→            }),
   722→          )),
   723→        );
   724→      }
   725→      entries.sort((a, b) => (b.modified ?? '').localeCompare(a.modified ?? ''));
   726→      return {
   727→        path,
   728→        total: names.length,
   729→        entries: entries.slice(0, listLimit),
   730→        ...(names.length > listLimit ? { truncated: true } : {}),
   731→      };
   732→    }
   733→    if (action === 'LOCAL_FILES_READ') {
   734→      const path = await target(input.path);
   735→      const { size, bytes } = await readFile(path, readBytes);
   736→      const truncated = size > readBytes;
   737→      if (bytes.includes(0)) return { path, size, binary: true };
   738→      return {
   739→        path,
   740→        size,
   741→        text: new TextDecoder().decode(bytes, { stream: truncated }),
   742→        ...(truncated ? { truncated } : {}),
   743→      };
   744→    }
   745→    if (action === 'LOCAL_FILES_IMPORT') {
   746→      const path = await target(input.path);
   747→      if (((await maybeLstat(path))?.size ?? 0) > fileBytes)
   748→        throw new IntegrationError(`Files up to ${fileBytes / 1024 / 1024} MB can be copied.`);
   749→      const { size, bytes } = await readFile(path, fileBytes);
   750→      const name = nodePath.basename(path);
   751→      const file: IntegrationFile = { name, mimeType: fileMimeType(name), bytes };
   752→      return { path, size, file };
   753→    }
   754→    if (action === 'LOCAL_FILES_CREATE_FOLDER') {
   755→      const path = await destination(input.path);
   756→      await mkdir(path).catch((error: NodeJS.ErrnoException) => {
   757→        throw error.code === 'EEXIST' ? taken(path) : error;
   758→      });
   759→      return { created: path };
   760→    }
   761→    if (action === 'LOCAL_FILES_WRITE' || action === 'LOCAL_FILES_EXPORT') {
   762→      const path = await destination(input.path);
   763→      const previous = await regularFile(path);
   764→      if (previous && !input.overwrite)
   765→        throw new IntegrationError(`${path} already exists. Set overwrite to replace it.`, 409);
   766→      let data: string | Uint8Array = input.text;
   767→      if (action === 'LOCAL_FILES_EXPORT') {
   768→        if (!context.read) throw new IntegrationError('Workspace files are unavailable.', 503);
   769→        data = await context.read(input.file);
   770→      }
   771→      const previousVersion = await replace(path, data, previous, keep, signal);
   772→      await quarantine(path);
   773→      return { written: path, ...(previousVersion ? { previousVersion } : {}) };
   774→    }
   775→    if (action === 'LOCAL_FILES_EDIT') {
   776→      const path = await target(input.path);
   777→      const info = await regularFile(path);
   778→      if (!info) throw new IntegrationError(`${path} does not exist.`, 404);
   779→      if (info.size > writeBytes)
   780→        throw new IntegrationError('Only files up to 1 MB can be edited.');
   781→      const { bytes } = await readFile(path, writeBytes);
   782→      let text: string;
   783→      try {
   784→        if (bytes.includes(0)) throw new Error();
   785→        text = new TextDecoder('utf-8', { fatal: true }).decode(bytes);
   786→      } catch {
   787→        throw new IntegrationError(`${path} is not a UTF-8 text file. Nothing was changed.`);
   788→      }
   789→      for (const { oldText, newText } of input.edits) {
   790→        const at = text.indexOf(oldText);
   791→        if (at < 0 || text.indexOf(oldText, at + 1) >= 0)
   792→          throw new IntegrationError(
   793→            `Each oldText must appear exactly once in ${path}. Nothing was changed.`,
   794→          );
   795→        text = text.slice(0, at) + newText + text.slice(at + oldText.length);
   796→      }
   797→      const previousVersion = await replace(path, text, info, keep, signal);
   798→      await quarantine(path);
   799→      return { edited: path, previousVersion };
   800→    }
   801→    if (action === 'LOCAL_FILES_MOVE') {
   802→      const { done, failed } = await each(
   803→        input.moves,
   804→        async ({ from, to }) => {
   805→          const checked = bound(await present(from, keep), from);
   806→          const moved = await destination(to, checked);
   807→          const source = bound(await present(from, keep), from);
   808→          await place(source, moved, keep);
   809→          return { from: source, to: moved };
   810→        },
   811→        signal,
   812→      );
   813→      return { moved: done, failed };
   814→    }
   815→    const { done, failed } = await each(
   816→      input.paths,
   817→      async (path) => {
   818→        const source = bound(await present(path, keep), path);
   819→        await toTrash(source, signal);
   820→        return source;
   821→      },
   822→      signal,
   823→    );
   824→    return { trashed: done, failed };
   825→  }

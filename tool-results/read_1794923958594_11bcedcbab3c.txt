    81→function chatConfirmationSql(match: string) {
    82→  return `SELECT a.*,u.source AS turn_source,u.status AS turn_status,u.background AS turn_background,u.session_id AS turn_session_id,u.prompt AS turn_prompt,to_jsonb(u)->>'task_id' AS turn_task_id
    83→  FROM approvals a JOIN turns u ON u.id=a.turn_id
    84→  WHERE ${match}
    85→    AND a.owner_id=$2 AND u.owner_id=$2 AND a.thread_id=$3 AND u.thread_id=$3
    86→    AND (u.channel=$4 OR a.confirmation_message_id=COALESCE((SELECT m.reply_to_message_id FROM messages m
    87→      WHERE m.id=$5::uuid AND m.owner_id=$2 AND m.thread_id=$3 AND m.role='assistant' AND m.turn_id IS NULL AND m.hidden_at IS NULL),$5::uuid))
    88→    AND EXISTS (SELECT 1 FROM messages c WHERE c.id=a.confirmation_message_id AND c.owner_id=a.owner_id AND c.hidden_at IS NULL)
    89→    AND (to_jsonb(u)->>'task_id' IS NULL OR NOT EXISTS (
    90→      SELECT 1 FROM turns n WHERE n.owner_id=u.owner_id AND n.session_id=u.session_id AND n.id<>u.id
    91→        AND n.source IN ('user','task') AND n.message_id IS NOT NULL AND n.status IN ('queued','running')))`;
    92→}

   113→function confirmedOutcome(
   114→  action: string,
   115→  { status, result, error }: { status: string; result?: unknown; error?: unknown },
   116→  resumed = false,
   117→) {
   118→  if (status === 'completed') {
   119→    const legacyPause = action === 'NOD_schedule_cancel';
   120→    const schedule = legacyPause
   121→      ? (result as {
   122→          kind?: string;
   123→          enabled?: boolean;
   124→          runAt?: string | null;
   125→          lastRun?: { scheduledFor?: string } | null;
   126→        })
   127→      : null;
   128→    const paused =
   129→      schedule?.enabled === false &&
   130→      (schedule.kind === 'recurring' ||
   131→        (schedule.kind === 'once' && schedule.lastRun?.scheduledFor !== schedule.runAt));
   132→    return `The exact action confirmed by the user was executed once by the trusted worker. Do not repeat it.${resumed ? ' Continue from its result.' : ''}${paused ? ' The schedule is paused, remains in Scheduled, and can be resumed.' : ''}\n${untrustedBlock(`${legacyPause ? 'NOD_schedule_pause' : action} result`, JSON.stringify(result))}`;
   133→  }
   134→  const failure =
   135→    error === undefined
   136→      ? ''
   137→      : `\n${untrustedBlock(`${action} error`, error instanceof Error ? error.message : String(error))}`;
   138→  return status === 'uncertain' || status === 'executing'
   139→    ? `The action the user confirmed failed or was interrupted and will not be retried. Its outcome is uncertain, so check the app before trying it again. Do not report it as done.${failure}`
   140→    : `The action the user confirmed did not run, so nothing was done. Do not report it as done.${failure}`;
   141→}

   143→async function turnRow(client: PoolClient, turnId: string) {
   144→  return (
   145→    await client.query<TurnRow>(
   146→      "SELECT id,thread_id,to_jsonb(turns)->>'task_id' AS task_id,source,parent_turn_id,message_id,prompt,background,channel FROM turns WHERE id=$1 AND owner_id=$2",
   147→      [turnId, ownerId],
   148→    )
   149→  ).rows[0];
   150→}
   151→
   152→async function turnChain(client: PoolClient, turn: TurnRow) {
   153→  const chain = [turn];
   154→  for (let current = turn; current.parent_turn_id;) {
   155→    const next = await turnRow(client, current.parent_turn_id);
   156→    if (!next || next.thread_id !== turn.thread_id || chain.some((item) => item.id === next.id))
   157→      return undefined;
   158→    chain.push(next);
   159→    current = next;
   160→  }
   161→  return chain;
   162→}

   164→async function scheduleAuthority(client: PoolClient, turnId: string) {
   165→  if (
   166→    !(await client.query("SELECT to_regclass('schedule_runs') AS table_name")).rows[0]?.table_name
   167→  )
   168→    return undefined;
   169→  return (
   170→    await client.query<{
   171→      authority: OwnerInstructions | null;
   172→      scheduled_for: Date;
   173→      time_zone: string;
   174→      title: string;
   175→      kind: string | null;
   176→    }>(
   177→      `SELECT to_jsonb(r)->'authority' AS authority,r.scheduled_for,r.time_zone,r.title,s.kind
   178→     FROM schedule_runs r LEFT JOIN schedules s ON s.id=r.schedule_id AND s.owner_id=r.owner_id
   179→     WHERE r.turn_id=$1 AND r.owner_id=$2`,
   180→      [turnId, ownerId],
   181→    )
   182→  ).rows[0];
   183→}

   185→async function workOf(
   186→  client: PoolClient,
   187→  taskId: string | null,
   188→  schedule: Awaited<ReturnType<typeof scheduleAuthority>>,
   189→): Promise<ApprovalWork | undefined> {
   190→  if (schedule) {
   191→    const ownerRequest = schedule.authority?.messages
   192→      .filter((message) => message.canAuthorize !== false)
   193→      .map((message) => message.text)
   194→      .filter(Boolean)
   195→      .join('\n\n');
   196→    return {
   197→      kind: 'schedule',
   198→      title: schedule.title,
   199→      ...(ownerRequest ? { ownerRequest } : {}),
   200→      ...(schedule.kind === 'recurring' ? { waitsForAnswer: true as const } : {}),
   201→    };
   202→  }
   203→  if (!taskId) return undefined;
   204→  const task = (
   205→    await client.query<{ title: string; request: string | null }>(
   206→      `SELECT t.title,m.text AS request FROM tasks t
   207→       LEFT JOIN messages m ON m.id=t.source_message_id AND m.owner_id=t.owner_id AND m.role='user' AND NOT ${withdrawnSql('m')}
   208→       WHERE t.id=$1 AND t.owner_id=$2`,
   209→      [taskId, ownerId],
   210→    )
   211→  ).rows[0];
   212→  return (
   213→    task && {
   214→      kind: 'task',
   215→      title: task.title,
   216→      ...(task.request ? { ownerRequest: task.request } : {}),
   217→    }
   218→  );
   219→}

   221→export async function turnOrigin(turnId: string) {
   222→  const client = await pool.connect();
   223→  try {
   224→    const current = await turnRow(client, turnId);
   225→    const chain = current && (await turnChain(client, current));
   226→    const origin = chain?.at(-1);
   227→    if (!chain || !origin) return undefined;
   228→    const run =
   229→      origin.source === 'schedule' ? await scheduleAuthority(client, origin.id) : undefined;
   230→    const task = current.task_id
   231→      ? (
   232→          await client.query<{ source_message_id: string | null }>(
   233→            'SELECT source_message_id FROM tasks WHERE id=$1 AND owner_id=$2',
   234→            [current.task_id, ownerId],
   235→          )
   236→        ).rows[0]
   237→      : undefined;
   238→    const said =
   239→      origin.source === 'user'
   240→        ? [
   241→            ...new Set([
   242→              ...chain.flatMap((turn) => (turn.message_id ? [turn.message_id] : [])),
   243→              ...(task?.source_message_id ? [task.source_message_id] : []),
   244→            ]),
   245→          ]
   246→        : [];
   247→    return {
   248→      taskId: current.task_id,
   249→      source: origin.source,
   250→      messageIds: said,
   251→      channel: current.channel,
   252→      scheduleInstruction: run?.authority ?? undefined,
   253→      ...(run
   254→        ? {
   255→            scheduledFor: localTimestamp(run.time_zone, run.scheduled_for),
   256→            timeZone: run.time_zone,
   257→          }
   258→        : {}),
   259→    };
   260→  } finally {
   261→    client.release();
   262→  }
   263→}

   265→export async function createApprovals(
   266→  execute: (
   267→    action: string,
   268→    args: Record<string, unknown>,
   269→    context: ApprovalExecutionContext,
   270→  ) => Promise<unknown>,
   271→  preflight?: (
   272→    action: string,
   273→    args: Record<string, unknown>,
   274→    expectedRevision?: number,
   275→    client?: PoolClient,
   276→  ) => Promise<void>,
   277→  interpret?: (reply: string) => Promise<boolean | undefined>,
   278→  describe?: (
   279→    proposal: ApprovalDescription,
   280→    signal?: AbortSignal,
   281→  ) => Promise<string | ApprovalPresentation>,
   282→  analytics?: CaptureAnalytics,
   283→) {
   284→  const changed = (
   285→    status: 'asked' | 'approved' | 'declined' | 'executed' | 'expired' | 'uncertain',
   286→    action: string,
   287→    source: string,
   288→  ) => {
   289→    if (!['user', 'schedule', 'event', 'resume', 'task'].includes(source)) return;
   290→    analytics?.({
   291→      event: 'confirmation_changed',
   292→      properties: {
   293→        status,
   294→        action_kind:
   295→          action === 'NOD_browser'
   296→            ? 'browser'
   297→            : action === 'NOD_vault_allow'
   298→              ? 'sign_in'
   299→              : action.startsWith('NOD_memory_')
   300→                ? 'memory'
   301→                : action.startsWith('NOD_')
   302→                  ? 'settings'
   303→                  : 'app',
   304→        source: source as TurnSource,
   305→      },
   306→    });
   307→  };
   308→  await pool.query(`${locksTable};
   309→    UPDATE approvals SET status='uncertain' WHERE status='executing';
   310→  `);

   312→  async function request(
   313→    turn: TurnRef,
   314→    action: string,
   315→    args: Record<string, unknown>,
   316→    options: {
   317→      allowed?: boolean;
   318→      run?: () => Promise<unknown>;
   319→      details?: Record<string, unknown>;
   320→      signal?: AbortSignal;
   321→      sourceMessageId?: string | null;
   322→      admit?: (asks: boolean) => Promise<void>;
   323→      validate?: (client: PoolClient) => Promise<void>;
   324→      expectedRevision?: number;
   325→      expectedChatRevision?: string;
   326→      basis?: TrustBasis;
   327→      instructionChange?: InstructionChange;
   328→    } = {},
   329→  ) {
   330→    const instructionChange = options.instructionChange
   331→      ? structuredClone(options.instructionChange)
   332→      : undefined;
   333→    const replacesInstruction =
   334→      action === 'NOD_schedule_create' ||
   335→      (action === 'NOD_schedule_update' && typeof instructionChange?.changes.prompt === 'string');
   336→    const encoded = JSON.stringify(args);
   337→    if (!encoded) throw new Error('The action has no arguments to save.');
   338→    const snapshot = JSON.parse(encoded) as Record<string, unknown>;
   339→    const details = options.details ? JSON.stringify(options.details) : undefined;
   340→    turn = { ...turn };
   341→    const allowed = options.allowed === true;
   342→    const expectedRevision = options.expectedRevision;
   343→    options.signal?.throwIfAborted();
   344→    await options.admit?.(!allowed);
   345→    const inspect = async (client: PoolClient, describing = false) => {
   346→      options.signal?.throwIfAborted();
   347→      const current = await turnRow(client, turn.turnId);
   348→      if (!current || current.thread_id !== turn.threadId)
   349→        throw conflict('This turn is no longer running.');
   350→      const chain = await turnChain(client, current);
   351→      const origin = chain?.at(-1);
   352→      if (!chain || !origin) throw new Error('This action needs a direct request from the user.');
   353→      let steering: { id: string; text: string } | undefined;
   354→      if (options.sourceMessageId && options.sourceMessageId !== current.message_id) {
   355→        if (!current.background)
   356→          steering = (
   357→            await client.query<{ id: string; text: string }>(
   358→              "SELECT id,text FROM messages WHERE id=$1 AND owner_id=$2 AND thread_id=$3 AND turn_id=$4 AND role='user'",
   359→              [options.sourceMessageId, ownerId, turn.threadId, turn.turnId],
   360→            )
   361→          ).rows[0];
   362→        if (!steering) throw new Error('This action needs a direct request from the user.');
   363→      }
   364→      const background = !steering && ['event', 'schedule'].includes(origin.source);
   365→      let authorMessageId: string | null;
   366→      const schedule =
   367→        origin.source === 'schedule' ? await scheduleAuthority(client, origin.id) : undefined;
   368→      if (schedule?.authority) await validateOwnerInstructions(client, ownerId, schedule.authority);
   369→      if (steering) {
   370→        authorMessageId = steering.id;
   371→      } else if (origin.source === 'event') {
   372→        authorMessageId = null;
   373→      } else if (origin.source === 'schedule') {
   374→        authorMessageId =
   375→          schedule?.authority?.messages.find(
   376→            (message) => message.canAuthorize !== false && message.id,
   377→          )?.id ?? null;
   378→      } else {
   379→        const author =
   380→          origin.source === 'user' && origin.message_id
   381→            ? (
   382→                await client.query<{ id: string; text: string }>(
   383→                  "SELECT id,text FROM messages WHERE id=$1 AND owner_id=$2 AND thread_id=$3 AND role='user'",
   384→                  [origin.message_id, ownerId, origin.thread_id],
   385→                )
   386→              ).rows[0]
   387→            : undefined;
   388→        if (!author) throw new Error('This action needs a direct request from the user.');
   389→        authorMessageId = author.id;
   390→      }
   391→      if (background) await lockKey(client, `approvals:${ownerId}:${turn.threadId}`);
   392→      let previous = (
   393→        background
   394→          ? await client.query(
   395→              "SELECT * FROM approvals WHERE turn_id=ANY($1::uuid[]) AND owner_id=$2 AND action=$3 AND args=$4::jsonb AND status<>'expired' ORDER BY id LIMIT 1",
   396→              [chain.map((item) => item.id), ownerId, action, snapshot],
   397→            )
   398→          : await client.query(
   399→              "SELECT * FROM approvals WHERE thread_id=$1 AND owner_id=$2 AND source_message_id=$3 AND action=$4 AND args=$5::jsonb AND status<>'expired' ORDER BY id LIMIT 1",
   400→              [turn.threadId, ownerId, authorMessageId, action, snapshot],
   401→            )
   402→      ).rows[0];
   403→      if (
   404→        previous?.status === 'pending' &&
   405→        !(
   406→          await client.query(
   407→            "SELECT 1 FROM turns WHERE id=$1 AND owner_id=$2 AND status IN ('running','waiting')",
   408→            [previous.turn_id, ownerId],
   409→          )
   410→        ).rowCount
   411→      ) {
   412→        await client.query("UPDATE approvals SET status='expired' WHERE id=$1 AND owner_id=$2", [
   413→          previous.id,
   414→          ownerId,
   415→        ]);
   416→        afterCommit(client, () => changed('expired', action, current.source));
   417→        previous = undefined;
   418→      }
   419→      if (previous) {
   420→        if (previous.status === 'completed')
   421→          return { cached: true as const, result: previous.result };
   422→        throw conflict('This exact action already has a pending or uncertain outcome.');
   423→      }
   424→      await preflight?.(action, snapshot, expectedRevision, client);
   425→      if (
   426→        !allowed &&
   427→        (
   428→          await client.query(
   429→            "SELECT 1 FROM approvals WHERE turn_id=$1 AND owner_id=$2 AND status='pending'",
   430→            [turn.turnId, ownerId],
   431→          )
   432→        ).rowCount
   433→      )
   434→        throw conflict('This turn is already waiting for an answer. Ask again after it.');
   435→      if (background) {
   436→        const pending = (
   437→          await client.query(
   438→            `SELECT a.*,m.text AS question FROM approvals a
   439→             JOIN turns u ON u.id=a.turn_id AND u.owner_id=a.owner_id AND u.thread_id=a.thread_id
   440→             JOIN messages m ON m.id=a.confirmation_message_id AND m.owner_id=a.owner_id AND m.thread_id=a.thread_id AND m.role='assistant'
   441→             WHERE a.owner_id=$1 AND a.thread_id=$2 AND a.action=$3 AND a.args=$4::jsonb
   442→               AND a.status='pending' AND u.status IN ('running','waiting') AND u.channel=$5
   443→               AND a.expected_revision IS NOT DISTINCT FROM $6::integer
   444→             ORDER BY a.id LIMIT 1`,
   445→            [ownerId, turn.threadId, action, snapshot, current.channel, expectedRevision ?? null],
   446→          )
   447→        ).rows[0];
   448→        if (pending)
   449→          return {
   450→            cached: false as const,
   451→            approval: pending,
   452→            question: pending.question as string,
   453→          };
   454→      }
   455→      return {
   456→        authorMessageId,
   457→        source: current.source,
   458→        ...(describing ? { work: await workOf(client, current.task_id, schedule) } : {}),
   459→      };
   460→    };
   461→    let question: string | undefined;
   462→    let title: string | undefined;
   463→    if (!allowed) {
   464→      const existing = await withTurnLock(turn.turnId, (client) => inspect(client, true));
   465→      if ('cached' in existing) {
   466→        if (existing.cached) return existing.result;
   467→        return { [appConfirmation]: existing.question as string };
   468→      }
   469→      if (!describe) throw new Error('The confirmation formatter is unavailable.');
   470→      const presentation = await describe(
   471→        {
   472→          turn: { ...turn },
   473→          action,
   474→          args: JSON.parse(encoded),
   475→          ...(details || (replacesInstruction && instructionChange)
   476→            ? {
   477→                details: {
   478→                  ...(details ? JSON.parse(details) : {}),
   479→                  ...(replacesInstruction && instructionChange
   480→                    ? { scheduleDefinition: instructionChange.after }
   481→                    : {}),
   482→                },
   483→              }
   484→            : {}),
   485→          ...(existing.work ? { work: existing.work } : {}),
   486→        },
   487→        options.signal,
   488→      );
   489→      question = typeof presentation === 'string' ? presentation : presentation.question;
   490→      title = typeof presentation === 'string' ? undefined : presentation.action;
   491→      options.signal?.throwIfAborted();
   492→      if (typeof question !== 'string' || !question.trim())
   493→        throw new Error('The confirmation formatter did not produce a valid question.');
   494→      if (title !== undefined && (!title.trim() || title.length > 64 || /[\r\n]/.test(title)))
   495→        throw new Error('The confirmation model did not produce a valid action label.');
   496→    }
   497→    const saveProposal = async (client: PoolClient) => {
   498→      if (
   499→        options.expectedChatRevision !== undefined &&
   500→        (await ownerChatRevision(client, ownerId, turn.threadId)) !== options.expectedChatRevision
   501→      )
   502→        throw Object.assign(
   503→          conflict(
   504→            'The chat changed while this action was being checked. This action did not run. Please ask again.',
   505→          ),
   506→          { notRun: true },
   507→        );
   508→      if (instructionChange) await validateInstructionChange(client, ownerId, instructionChange);
   509→      const inspected = await inspect(client);
   510→      if ('cached' in inspected) return inspected;
   511→      await options.validate?.(client);
   512→      options.signal?.throwIfAborted();
   513→      if (!question) return { cached: false as const, authorMessageId: inspected.authorMessageId };
   514→      const messageId = randomUUID();
   515→      await client.query(
   516→        "INSERT INTO messages(id,owner_id,thread_id,turn_id,role,text) VALUES($1,$2,$3,$4,'assistant',$5)",
   517→        [messageId, ownerId, turn.threadId, turn.turnId, question],
   518→      );
   519→      const approval = (
   520→        await client.query(
   521→          "INSERT INTO approvals(id,owner_id,thread_id,turn_id,action,args,status,source_message_id,confirmation_message_id,expected_revision,details,title,instruction_change) VALUES($1,$2,$3,$4,$5,$6,'pending',$7,$8,$9,$10::jsonb,$11,$12::jsonb) RETURNING *",
   522→          [
   523→            randomUUID(),
   524→            ownerId,
   525→            turn.threadId,
   526→            turn.turnId,
   527→            action,
   528→            snapshot,
   529→            inspected.authorMessageId,
   530→            messageId,
   531→            expectedRevision ?? null,
   532→            details ?? null,
   533→            title ?? null,
   534→            instructionChange ? JSON.stringify(instructionChange) : null,
   535→          ],
   536→        )
   537→      ).rows[0];
   538→      options.signal?.throwIfAborted();
   539→      return { cached: false as const, approval, question, source: inspected.source };
   540→    };
   541→    const proposal =
   542→      options.expectedChatRevision !== undefined
   543→        ? await transaction(
   544→            async (client) => {
   545→              if ((await lockThread(client)).id !== turn.threadId)
   546→                throw conflict('This conversation is no longer active.');
   547→              return withTurnLock(turn.turnId, saveProposal, client);
   548→            },
   549→            pool,
   550→            options.signal,
   551→          )
   552→        : await withTurnLock(turn.turnId, saveProposal);
   553→    if (proposal.cached) return proposal.result;
   554→    if ('question' in proposal) {
   555→      if ('source' in proposal && proposal.source) changed('asked', action, proposal.source);
   556→      refresh();
   557→      return { [appConfirmation]: proposal.question as string };
   558→    }
   559→    if (options.run) return options.run();
   560→    return execute(action, snapshot, {
   561→      threadId: turn.threadId,
   562→      turnId: turn.turnId,
   563→      sourceMessageId: proposal.authorMessageId,
   564→      signal: options.signal,
   565→      ...(options.details ? { details: options.details } : {}),
   566→      ...(expectedRevision === undefined ? {} : { expectedRevision }),
   567→      ...(allowed && options.basis ? { basis: options.basis } : {}),
   568→      ...(instructionChange ? { instructionChange: instructionChange } : {}),
   569→    });
   570→  }

   572→  async function confirmationFor(input: ChatConfirmationInput) {
   573→    const target =
   574→      input.replyToMessageId ??
   575→      (
   576→        await pool.query(
   577→          `SELECT m.id FROM messages m JOIN turns u ON u.id=m.turn_id WHERE m.owner_id=$1 AND u.owner_id=$1 AND m.thread_id=$2 AND u.channel=$3 AND m.role='assistant' AND m.hidden_at IS NULL
   578→           AND NOT EXISTS (SELECT 1 FROM messages s WHERE s.id=$4 AND s.owner_id=$1 AND s.thread_id=$2 AND date_trunc('milliseconds',s.created_at)<date_trunc('milliseconds',m.created_at))
   579→           AND NOT EXISTS (SELECT 1 FROM messages n WHERE n.owner_id=$1 AND n.thread_id=$2 AND n.role='assistant' AND n.turn_id IS NULL AND n.delivered_channel=$3 AND n.hidden_at IS NULL
   580→             AND (n.created_at,n.id)>(m.created_at,m.id)
   581→             AND NOT EXISTS (SELECT 1 FROM messages s WHERE s.id=$4 AND s.owner_id=$1 AND s.thread_id=$2 AND date_trunc('milliseconds',s.created_at)<date_trunc('milliseconds',n.created_at)))
   582→           ORDER BY m.created_at DESC,m.id DESC LIMIT 1`,
   583→          [ownerId, input.threadId, input.channel, input.seenMessageId ?? null],
   584→        )
   585→      ).rows[0]?.id;
   586→    if (!target) return undefined;
   587→    return (
   588→      await pool.query(
   589→        chatConfirmationSql(`a.confirmation_message_id=COALESCE((SELECT m.reply_to_message_id FROM messages m
   590→         WHERE m.id=$1 AND m.owner_id=$2 AND m.thread_id=$3 AND m.role='assistant' AND m.turn_id IS NULL AND m.hidden_at IS NULL),$1)`),
   591→        [target, ownerId, input.threadId, input.channel, input.replyToMessageId ?? null],
   592→      )
   593→    ).rows[0];
   594→  }

   596→  async function answerChat(input: ChatConfirmationInput) {
   597→    if (!interpret) return undefined;
   598→    const approval = await confirmationFor(input);
   599→    if (!approval) return undefined;
   600→    const approvalId = approval.id as string;
   601→    if (input.hasAttachments) return { approvalId };
   602→    try {
   603→      const approved = await interpret(input.text);
   604→      return approved === undefined ? { approvalId } : { approvalId, approved };
   605→    } catch {
   606→      return undefined;
   607→    }
   608→  }

   610→  async function acceptChat(
   611→    client: PoolClient,
   612→    input: ChatConfirmationInput,
   613→    onInserted?: () => void,
   614→  ): Promise<ChatConfirmationResult> {
   615→    if (!input.answer) return undefined;
   616→    const approval = (
   617→      await client.query(
   618→        `${chatConfirmationSql(input.clicked ? 'a.id=$1 AND a.confirmation_message_id=$5' : 'a.id=$1')} FOR UPDATE OF a`,
   619→        [
   620→          input.answer.approvalId,
   621→          ownerId,
   622→          input.threadId,
   623→          input.channel,
   624→          input.replyToMessageId ?? null,
   625→        ],
   626→      )
   627→    ).rows[0];
   628→    if (approval?.status !== 'pending') return undefined;
   629→    const { approved } = input.answer;
   630→    const active = ['running', 'waiting'].includes(approval.turn_status);
   631→    if (input.clicked && (approved === undefined || !active)) return undefined;
   632→    if (approved === undefined) {
   633→      await client.query("UPDATE approvals SET status='expired' WHERE id=$1 AND owner_id=$2", [
   634→        approval.id,
   635→        ownerId,
   636→      ]);
   637→      afterCommit(client, () => changed('expired', approval.action, approval.turn_source));
   638→      if (active && approval.turn_background && input.replyToMessageId)
   639→        return {
   640→          waiting: {
   641→            id: approval.turn_id as string,
   642→            session_id: approval.turn_session_id as string,
   643→            background: true,
   644→          },
   645→        };
   646→      const cancelled = await client.query(
   647→        "UPDATE turns SET status='cancelled',finished_at=now() WHERE id=$1 AND owner_id=$2 AND status='waiting' RETURNING id",
   648→        [approval.turn_id, ownerId],
   649→      );
   650→      if (cancelled.rowCount && approval.turn_background && approval.turn_task_id) {
   651→        const task = (
   652→          await client.query<{ id: string }>(
   653→            "SELECT id FROM tasks WHERE id=$1 AND owner_id=$2 AND turn_id=$3 AND state='open' FOR UPDATE",
   654→            [approval.turn_task_id, ownerId, approval.turn_id],
   655→          )
   656→        ).rows[0];
   657→        if (task) {
   658→          const messageId = randomUUID();
   659→          await client.query(
   660→            `INSERT INTO messages(id,owner_id,thread_id,turn_id,role,text,created_at)
   661→             VALUES($1,$2,$3,$4,'assistant',$5,clock_timestamp())`,
   662→            [
   663→              messageId,
   664→              ownerId,
   665→              input.threadId,
   666→              approval.turn_id,
   667→              'The proposed action did not run. This task is still open, but its current work stopped. Reply here to continue the remaining work.',
   668→            ],
   669→          );
   670→          await client.query(
   671→            'UPDATE tasks SET continuation_message_id=$3,updated_at=now() WHERE id=$1 AND owner_id=$2',
   672→            [task.id, ownerId, messageId],
   673→          );
   674→        }
   675→      }
   676→      return undefined;
   677→    }
   678→    if (!active) return undefined;
   679→    const prompt = approved
   680→      ? 'The user confirmed the exact saved action. The trusted worker must execute that snapshot once before continuing. Do not substitute an action or execute it yourself.'
   681→      : 'The user declined the saved action. Do not execute it or ask for it again. Continue only other already-authorized work, or finish quietly.';
   682→    const background = Boolean(approval.turn_background);
   683→    const turnId = await enqueueTurn(
   684→      background ? `${prompt}\nThe request was: ${approval.turn_prompt}` : prompt,
   685→      {
   686→        source: 'resume',
   687→        sourceId:
   688→          input.sourceId ?? `confirmation:${approval.id}:${approved ? 'accepted' : 'declined'}`,
   689→        channel: input.channel,
   690→        background,
   691→        sessionId: approval.turn_session_id,
   692→        parentTurnId: approval.turn_id,
   693→        client,
   694→      },
   695→    );
   696→    const messageId = input.clientMessageId ?? randomUUID();
   697→    await client.query('UPDATE turns SET message_id=$2 WHERE id=$1 AND owner_id=$3', [
   698→      turnId,
   699→      messageId,
   700→      ownerId,
   701→    ]);
   702→    await client.query(
   703→      "INSERT INTO messages(id,owner_id,role,text,display_text,thread_id,turn_id,reply_to_message_id,approval_answer) VALUES($1,$2,'user',$3,$4,$5,$6,$7,$8)",
   704→      [
   705→        messageId,
   706→        ownerId,
   707→        input.text,
   708→        input.displayText ?? null,
   709→        input.threadId,
   710→        turnId,
   711→        input.replyToMessageId ?? null,
   712→        input.clicked ? approved : null,
   713→      ],
   714→    );
   715→    await client.query('UPDATE approvals SET status=$2,execution_turn_id=$3 WHERE id=$1', [
   716→      approval.id,
   717→      approved ? 'confirmed' : 'declined',
   718→      turnId,
   719→    ]);
   720→    afterCommit(client, () =>
   721→      changed(approved ? 'approved' : 'declined', approval.action, approval.turn_source),
   722→    );
   723→    onInserted?.();
   724→    return { threadId: input.threadId, turnId, messageId };
   725→  }

   727→  async function executeConfirmed(turnId: string, signal: AbortSignal) {
   728→    signal.throwIfAborted();
   729→    const approval = await withTurnLock(turnId, async (client) => {
   730→      const selected = (
   731→        await client.query(
   732→          "SELECT a.*,u.source AS turn_source FROM approvals a JOIN turns u ON u.id=a.turn_id WHERE a.execution_turn_id=$1 AND a.owner_id=$2 AND a.status='confirmed' FOR UPDATE OF a",
   733→          [turnId, ownerId],
   734→        )
   735→      ).rows[0];
   736→      if (!selected) return undefined;
   737→      const refused = await (async () => {
   738→        if (selected.instruction_change)
   739→          await validateInstructionChange(client, ownerId, selected.instruction_change);
   740→        return preflight?.(
   741→          selected.action,
   742→          selected.args,
   743→          selected.expected_revision ?? undefined,
   744→          client,
   745→        );
   746→      })().then(
   747→        () => undefined,
   748→        (error: unknown) => ({ error }),
   749→      );
   750→      signal.throwIfAborted();
   751→      await client.query('UPDATE approvals SET status=$2 WHERE id=$1', [
   752→        selected.id,
   753→        refused ? 'expired' : 'executing',
   754→      ]);
   755→      return { ...selected, refused };
   756→    });
   757→    if (!approval) return undefined;
   758→    const outcome: {
   759→      status: 'completed' | 'expired' | 'uncertain';
   760→      result?: unknown;
   761→      error?: unknown;
   762→    } = approval.refused
   763→      ? { status: 'expired', error: approval.refused.error }
   764→      : await execute(approval.action, approval.args, {
   765→          threadId: approval.thread_id,
   766→          turnId,
   767→          sourceMessageId: approval.source_message_id,
   768→          signal,
   769→          ...(approval.details ? { details: approval.details } : {}),
   770→          ...(approval.instruction_change
   771→            ? {
   772→                instructionChange: {
   773→                  ...approval.instruction_change,
   774→                  replaceInstruction:
   775→                    approval.action === 'NOD_schedule_create' ||
   776→                    (approval.action === 'NOD_schedule_update' &&
   777→                      typeof approval.instruction_change.changes.prompt === 'string'),
   778→                  evidence:
   779→                    approval.action === 'NOD_schedule_create' ||
   780→                    (approval.action === 'NOD_schedule_update' &&
   781→                      typeof approval.instruction_change.changes.prompt === 'string')
   782→                      ? {
   783→                          messages: [],
   784→                          approvals: [
   785→                            {
   786→                              id: approval.id,
   787→                              at: new Date().toISOString(),
   788→                              definition: approval.instruction_change.after,
   789→                            },
   790→                          ],
   791→                        }
   792→                      : {
   793→                          messages: [
   794→                            {
   795→                              id: null,
   796→                              approvalId: approval.id,
   797→                              at: new Date().toISOString(),
   798→                              text: '',
   799→                              fields: approval.instruction_change.changes,
   800→                            },
   801→                          ],
   802→                        },
   803→                },
   804→              }
   805→            : {}),
   806→          ...(approval.expected_revision === null
   807→            ? {}
   808→            : { expectedRevision: approval.expected_revision }),
   809→        })
   810→          .then((result) => ({
   811→            status: 'completed' as const,
   812→            result: JSON.parse(JSON.stringify(result ?? null)),
   813→          }))
   814→          .catch((error: unknown) => ({
   815→            status: notRun(error) ? ('expired' as const) : ('uncertain' as const),
   816→            error,
   817→          }));
   818→    if (!approval.refused)
   819→      await pool.query(
   820→        "UPDATE approvals SET status=$2,result=$3 WHERE id=$1 AND owner_id=$4 AND status='executing'",
   821→        [
   822→          approval.id,
   823→          outcome.status,
   824→          outcome.status === 'completed' ? JSON.stringify(outcome.result) : null,
   825→          ownerId,
   826→        ],
   827→      );
   828→    changed(
   829→      outcome.status === 'completed' ? 'executed' : outcome.status,
   830→      approval.action,
   831→      approval.turn_source,
   832→    );
   833→    signal.throwIfAborted();
   834→    return { prompt: confirmedOutcome(approval.action, outcome, true) };
   835→  }

   837→  async function stoppedOutcomes(turnId: string) {
   838→    const { rows } = await pool.query<{ action: string; status: string; result: unknown }>(
   839→      `WITH t AS (SELECT session_id,created_at,id FROM turns WHERE id=$1 AND owner_id=$2),
   840→       finished AS (SELECT d.created_at,d.id FROM turns d JOIN t ON d.session_id=t.session_id
   841→         WHERE d.owner_id=$2 AND d.status IN ('succeeded','waiting') AND (d.created_at,d.id)<(t.created_at,t.id)
   842→         ORDER BY d.created_at DESC,d.id DESC LIMIT 1)
   843→       SELECT a.action,a.status,a.result FROM t
   844→       JOIN turns e ON e.owner_id=$2 AND e.session_id=t.session_id AND e.status IN ('cancelled','failed')
   845→         AND (e.created_at,e.id)<(t.created_at,t.id)
   846→         AND NOT EXISTS (SELECT 1 FROM finished f WHERE (f.created_at,f.id)>(e.created_at,e.id))
   847→       JOIN approvals a ON a.owner_id=$2 AND a.execution_turn_id=e.id AND a.status<>'declined'
   848→       ORDER BY e.created_at,e.id`,
   849→      [turnId, ownerId],
   850→    );
   851→    return rows.map(({ action, ...outcome }) => confirmedOutcome(action, outcome));
   852→  }

   854→  async function retireScheduleConfirmationsInTransaction(
   855→    client: PoolClient,
   856→    id: string,
   857→    waitForSend: boolean,
   858→  ) {
   859→    const pending = (
   860→      await client.query<{
   861→        id: string;
   862→        action: string;
   863→        turn_id: string;
   864→        confirmation_message_id: string | null;
   865→        source: string;
   866→      }>(
   867→        `SELECT a.id,a.action,a.turn_id,a.confirmation_message_id,u.source
   868→         FROM approvals a JOIN turns u ON u.id=a.turn_id AND u.owner_id=a.owner_id
   869→         WHERE a.owner_id=$1 AND a.action IN ('NOD_schedule_pause','NOD_schedule_cancel','NOD_schedule_update')
   870→           AND lower(a.args->>'id')=lower($2::text) AND a.status='pending'
   871→         FOR UPDATE OF a`,
   872→        [ownerId, id],
   873→      )
   874→    ).rows;
   875→    if (!pending.length) return;
   876→    const approvalIds = pending.map((row) => row.id);
   877→    const turnIds = [...new Set(pending.map((row) => row.turn_id))];
   878→    const messageIds = pending.flatMap((row) =>
   879→      row.confirmation_message_id ? [row.confirmation_message_id] : [],
   880→    );
   881→    if (messageIds.length)
   882→      try {
   883→        if (waitForSend) await client.query("SET LOCAL lock_timeout = '30s'");
   884→        await client.query(
   885→          `SELECT id FROM messages WHERE owner_id=$1 AND id=ANY($2::uuid[]) ORDER BY id FOR UPDATE${waitForSend ? '' : ' NOWAIT'}`,
   886→          [ownerId, messageIds],
   887→        );
   888→      } catch (cause) {
   889→        if ((cause as { code?: string }).code !== '55P03') throw cause;
   890→        throw Object.assign(new Error('A schedule confirmation is still being sent.'), {
   891→          code: '55P03',
   892→          notRun: true,
   893→          statusCode: 409,
   894→        });
   895→      }
   896→    await client.query(
   897→      'SELECT id FROM turns WHERE owner_id=$1 AND id=ANY($2::uuid[]) ORDER BY id FOR NO KEY UPDATE NOWAIT',
   898→      [ownerId, turnIds],
   899→    );
   900→    await client.query(
   901→      "UPDATE approvals SET status='expired' WHERE owner_id=$1 AND id=ANY($2::uuid[])",
   902→      [ownerId, approvalIds],
   903→    );
   904→    if (messageIds.length)
   905→      await client.query(
   906→        'UPDATE messages SET hidden_at=now() WHERE owner_id=$1 AND id=ANY($2::uuid[])',
   907→        [ownerId, messageIds],
   908→      );
   909→    let tasks: Array<{ id: string; turn_id: string; thread_id: string }>;
   910→    try {
   911→      tasks = (
   912→        await client.query<{ id: string; turn_id: string; thread_id: string }>(
   913→          `SELECT t.id,t.turn_id,u.thread_id FROM tasks t
   914→           JOIN turns u ON u.id=t.turn_id AND u.owner_id=t.owner_id AND u.task_id=t.id
   915→           WHERE t.owner_id=$1 AND t.state='open' AND u.id=ANY($2::uuid[])
   916→           ORDER BY t.id FOR UPDATE OF t NOWAIT`,
   917→          [ownerId, turnIds],
   918→        )
   919→      ).rows;
   920→    } catch (cause) {
   921→      if ((cause as { code?: string }).code !== '55P03') throw cause;
   922→      throw Object.assign(new Error('This task is still being updated. Try again.'), {
   923→        code: '55P03',
   924→        notRun: true,
   925→        statusCode: 409,
   926→      });
   927→    }
   928→    for (const task of tasks) {
   929→      const messageId = randomUUID();
   930→      await client.query(
   931→        `INSERT INTO messages(id,owner_id,thread_id,turn_id,role,text)
   932→         VALUES($1,$2,$3,$4,'assistant',$5)`,
   933→        [
   934→          messageId,
   935→          ownerId,
   936→          task.thread_id,
   937→          task.turn_id,
   938→          'The schedule has changed. This task is still open. Reply here if you want to continue it.',
   939→        ],
   940→      );
   941→      await client.query(
   942→        'UPDATE tasks SET continuation_message_id=$3,updated_at=now() WHERE id=$1 AND owner_id=$2',
   943→        [task.id, ownerId, messageId],
   944→      );
   945→    }
   946→    await client.query(
   947→      "UPDATE turns SET status='cancelled',finished_at=now() WHERE owner_id=$1 AND id=ANY($2::uuid[]) AND status IN ('running','waiting')",
   948→      [ownerId, turnIds],
   949→    );
   950→    afterCommit(client, () => {
   951→      abortTurns(turnIds);
   952→      for (const row of pending) changed('expired', row.action, row.source);
   953→    });
   954→  }

   956→  async function retireScheduleConfirmations(client: PoolClient, id: string, waitForSend: boolean) {
   957→    try {
   958→      await retireScheduleConfirmationsInTransaction(client, id, waitForSend);
   959→    } catch (cause) {
   960→      const error = cause as { code?: string; statusCode?: number };
   961→      if (error?.code !== '55P03' || error.statusCode === 409) throw cause;
   962→      throw Object.assign(new Error('This schedule or task is still being updated. Try again.'), {
   963→        code: '55P03',
   964→        notRun: true,
   965→        statusCode: 409,
   966→      });
   967→    }
   968→  }

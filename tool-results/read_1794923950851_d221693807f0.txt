   100→   100→type ChoiceAnswerRow = { answer: unknown; text: string };

   102→   102→function approvalState(row: ApprovalRow): ApprovalState {
   103→   103→  if (row.status === 'pending') return row.active ? 'pending' : 'expired';
   104→   104→  if (row.status === 'confirmed') return row.executionActive ? 'working' : 'expired';
   105→   105→  if (row.status === 'executing') return row.executionActive ? 'working' : 'uncertain';
   106→   106→  if (row.status === 'completed') return 'done';
   107→   107→  if (row.status === 'declined' || row.status === 'uncertain') return row.status;
   108→   108→  return 'expired';
   109→   109→}
   110→   110→
   111→   111→function choiceAnswerView(options: AnswerChoice[], row: ChoiceAnswerRow | null | undefined) {
   112→   112→  const answer = questionAnswerSchema.safeParse(row?.answer);
   113→   113→  if (!row || !answer.success) return undefined;
   114→   114→  if (!('choiceIds' in answer.data)) return { choiceIds: [], text: row.text };
   115→   115→  const { choiceIds, custom } = answer.data;
   116→   116→  const text =
   117→   117→    custom &&
   118→   118→    writtenAnswer(
   119→   119→      options.filter((option) => choiceIds.includes(option.id)),
   120→   120→      row.text,
   121→   121→    );
   122→   122→  return { choiceIds, ...(text ? { text } : {}) };
   123→   123→}

   133→   133→function messageView(row: MessageRow): Message {
   134→   134→  const {
   135→   135→    assistantReaction,
   136→   136→    presentation,
   137→   137→    displayText,
   138→   138→    audioCaption,
   139→   139→    answerChoices,
   140→   140→    answerChoicesPending,
   141→   141→    approval,
   142→   142→    choiceAnswer,
   143→   143→    cardAnswer,
   144→   144→    work,
   145→   145→    ...message
   146→   146→  } = row;
   147→   147→  const visual =
   148→   148→    message.role === 'assistant' ? presentationSchema.safeParse(presentation) : undefined;
   149→   149→  const choices =
   150→   150→    message.role === 'assistant' ? answerChoicesSchema.safeParse(answerChoices) : undefined;
   151→   151→  const answered = choices?.success
   152→   152→    ? choiceAnswerView(choices.data.options, choiceAnswer)
   153→   153→    : undefined;
   154→   154→  return {
   155→   155→    ...message,
   156→   156→    ...(message.role === 'user' && typeof displayText === 'string' ? { displayText } : {}),
   157→   157→    ...(typeof audioCaption === 'string' ? { audioCaption } : {}),
   158→   158→    ...(visual?.success ? { presentation: visual.data } : {}),
   159→   159→    ...(choices?.success
   160→   160→      ? { answerChoices: choices.data, answerChoicesPending: answerChoicesPending === true }
   161→   161→      : {}),
   162→   162→    ...(answered ? { choiceAnswer: answered } : {}),
   163→   163→    ...(approval
   164→   164→      ? {
   165→   165→          approval: {
   166→   166→            id: approval.id,
   167→   167→            state: approvalState(approval),
   168→   168→            ...approvalTitle(approval.action),
   169→   169→            ...(approval.title ? { title: approval.title } : {}),
   170→   170→            ...approvalContent(approval.action, approval.args),
   171→   171→          },
   172→   172→        }
   173→   173→      : {}),
   174→   174→    ...(cardAnswer ? { cardAnswer: true as const } : {}),
   175→   175→    ...(work ? { work } : {}),
   176→   176→    reactions: [
   177→   177→      ...(message.reaction ? [{ actor: 'user' as const, reaction: message.reaction }] : []),
   178→   178→      ...(assistantReaction ? [{ actor: 'assistant' as const, reaction: assistantReaction }] : []),
   179→   179→    ],
   180→   180→  };
   181→   181→}

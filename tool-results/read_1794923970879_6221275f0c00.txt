   128→export function redactBrowserText(text: string, secrets: Iterable<string>): string {
   129→  const variants = new Set<string>();
   130→  for (const secret of secrets) {
   131→    if (!secret) continue;
   132→    variants.add(secret);
   133→    variants.add(JSON.stringify(secret).slice(1, -1));
   134→    try {
   135→      variants.add(encodeURIComponent(secret));
   136→      variants.add(encodeURI(secret));
   137→      variants.add(new URLSearchParams({ value: secret }).toString().slice(6));
   138→    } catch {}
   139→  }
   140→  for (const value of variants)
   141→    variants.add(value.replace(/%[0-9A-F]{2}/g, (code) => code.toLowerCase()));
   142→  for (const secret of secrets)
   143→    if (/^\d{12,19}$/.test(secret))
   144→      text = text.replace(new RegExp(secret.split('').join('\\D{0,3}'), 'g'), secret);
   145→  let parts: Array<string | null> = [text];
   146→  for (const value of [...variants].sort((a, b) => b.length - a.length))
   147→    parts = parts.flatMap((part) => {
   148→      if (part === null) return [null];
   149→      return part.split(value).flatMap((section, index) => (index ? [null, section] : [section]));
   150→    });
   151→  return parts.map((part) => part ?? '[redacted]').join('');
   152→}

   176→function visualGesture(input: BrowserAction) {
   177→  if (input.x === undefined && input.y === undefined && input.points === undefined) {
   178→    if (input.durationMs !== undefined) throw new Error('A pointer duration needs coordinates.');
   179→    return;
   180→  }

   182→    throw new Error('Choose either a selector or screenshot coordinates.');

   184→    input.action === 'click' && input.points === undefined
   185→      ? [browserPoint.parse({ x: input.x, y: input.y })]

   187→        ? z.array(browserPoint).min(2).parse(input.points)

   191→    .number()

   198→export function redactBrowserUrl(value: string, secrets: Iterable<string>): string {
   199→  const url = new URL(value);
   200→  url.username = '';
   201→  url.password = '';
   202→  const sensitive =
   203→    /^(?:token|accesstoken|idtoken|refreshtoken|password|otp|session|sessionid|ticket|assertion|authorization|apikey|xapikey|clientsecret|signature|sig|xamzsignature|xamzcredential|xgoogsignature|xgoogcredential)$/i;
   204→  const secret = (name: string) => sensitive.test(name.replace(/[-_]/g, ''));
   205→  for (const name of url.searchParams.keys())
   206→    if (secret(name)) url.searchParams.set(name, '[redacted]');
   207→  if (
   208→    url.hash
   209→      .slice(1)
   210→      .split(/[?&]/)
   211→      .some((part) => {
   212→        if (!part.includes('=')) return false;
   213→        try {
   214→          return secret(decodeURIComponent(part.split('=')[0]!));
   215→        } catch {
   216→          return true;
   217→        }
   218→      })
   219→  )
   220→    url.hash = '[redacted]';
   221→  return redactBrowserText(url.href, secrets);
   222→}

   378→  async function reserveBrowser(state: BrowserState, signal?: AbortSignal): Promise<void> {
   379→    const stopped = new Promise<never>((_, reject) => {
   380→      if (signal?.aborted) reject(signal.reason);
   381→      signal?.addEventListener('abort', () => reject(signal.reason), { once: true });
   382→    });
   383→    stopped.catch(() => {});
   384→    const reserve = () =>
   385→      allocate(async () => {
   386→        if (state.started || state.saved.navigationFailed) return true;
   387→        if (signal?.aborted) return false;
   388→        const occupied = () =>
   389→          [...browsers.values()].filter((entry) => entry.started || entry.saved.navigationFailed);
   390→        const attempted = new Set<string>();
   391→        while (occupied().length >= maxLiveSessions) {
   392→          const candidate = occupied()
   393→            .filter(
   394→              (entry) =>
   395→                entry !== state &&
   396→                !attempted.has(entry.id) &&
   397→                !entry.busy &&
   398→                !entry.closed &&
   399→                !entry.closing &&
   400→                !entry.injecting &&
   401→                !holding(entry) &&
   402→                Date.now() - entry.lastUsed >= evictIdleMs,
   403→            )
   404→            .sort((a, b) => a.lastUsed - b.lastUsed)[0];
   405→          if (!candidate) return false;
   406→          attempted.add(candidate.id);
   407→          try {
   408→            await closeBrowser(candidate.id, candidate.lastUsed);
   409→          } catch {
   410→            throw new Error(
   411→              'The browser sessions are busy. Finish an open browser task and retry.',
   412→            );
   413→          }
   414→        }
   415→        if (disposed || state.closed || state.closing)
   416→          throw new Error('The browser session has expired. Open the URL again.');
   417→        if (signal?.aborted) return false;
   418→        state.started = true;
   419→        return true;
   420→      });
   421→    while (!(await Promise.race([reserve(), stopped]))) {
   422→      if (disposed || state.closed || state.closing)
   423→        throw new Error('The browser session has expired. Open the URL again.');
   424→      await delay(500, undefined, { signal });
   425→    }
   426→  }

   445→  async function kill(state: BrowserState) {
   446→    const socket = state.socket ?? (await daemonSocket(state).catch(() => undefined));
   447→    const pidFile = socket?.replace(/\.sock$/, '.pid');
   448→    const host = `${hostname()}-`;
   449→    const profile = profilePath(state);
   450→    const lock = await readlink(resolve(profile, 'SingletonLock')).catch(() => '');
   451→    const owned = async (pid: number, file: string, marker: string) =>
   452→      Number.isInteger(pid) &&
   453→      pid > 1 &&
   454→      pid !== process.pid &&
   455→      (process.platform !== 'linux' ||
   456→        (await readFile(`/proc/${pid}/${file}`, 'utf8').catch(() => '')).includes(marker));
   457→    const chrome = lock.startsWith(host) ? Number(lock.slice(host.length)) : 0;
   458→    const daemon = Number(pidFile ? await readFile(pidFile, 'utf8').catch(() => '') : '');
   459→    const targets = [
   460→      ...((await owned(chrome, 'cmdline', `--user-data-dir=${profile}\0`)) ? [chrome] : []),
   461→      ...((await owned(daemon, 'environ', `AGENT_BROWSER_SESSION=${browserSessionId(state.id)}\0`))
   462→        ? [daemon]
   463→        : []),
   464→    ];
   465→    const signal = (pid: number, value: NodeJS.Signals | 0) => {
   466→      try {
   467→        process.kill(pid, value);
   468→        return true;
   469→      } catch {
   470→        return false;
   471→      }
   472→    };
   473→    for (const pid of targets) {
   474→      signal(-pid, 'SIGKILL');
   475→      signal(pid, 'SIGKILL');
   476→    }
   477→    for (let waited = 0; waited < 5000 && targets.some((pid) => signal(pid, 0)); waited += 50)
   478→      await delay(50);
   479→    await Promise.all([socket, pidFile].map((path) => path && rm(path, { force: true })));
   480→    delete state.socket;
   481→  }

   510→  async function command(
   511→    state: BrowserState,
   512→    request: Record<string, unknown>,
   513→    options: { signal?: AbortSignal; maxBytes?: number; timeoutMs?: number } = {},
   514→  ): Promise<Record<string, unknown>> {
   515→    options.signal?.throwIfAborted();
   516→    const action = String(request.action);
   517→    if (state.socket) {
   518→      const result = await drainInterruptedCommand(state, action, () =>
   519→        requestDaemon(state.socket!, request, options.signal, options.timeoutMs, options.maxBytes),
   520→      );
   521→      if (action === 'close') {
   522→        const started = Date.now();
   523→        while (await stat(state.socket!).catch(() => null)) {
   524→          if (Date.now() - started > 5000) throw new Error('The previous browser did not stop.');
   525→          await delay(50);
   526→        }
   527→      }
   528→      return result;
   529→    }
   530→    if (action !== 'navigate' && action !== 'close')
   531→      throw new Error('Reopen this page before continuing browser actions.');
   532→    await mkdir(runtimeRoot, { recursive: true, mode: 0o700 });
   533→    const config = resolve(runtimeRoot, 'browser-config.json');
   534→    await writeFile(config, '{}', { mode: 0o600 });
   535→    const initScript = resolve(runtimeRoot, 'browser-init.js');
   536→    await writeFile(initScript, withoutPasskeys, { mode: 0o600 });
   537→    state.proxy ??= createBrowserProxy({ privateNetwork });
   538→    const proxyUrl = await state.proxy.url();
   539→    const profile = profilePath(state);
   540→    const lockHost = /^(.+)-\d+$/.exec(
   541→      await readlink(resolve(profile, 'SingletonLock')).catch(() => ''),
   542→    )?.[1];
   543→    if (lockHost !== undefined && lockHost !== hostname())
   544→      await rm(resolve(profile, 'SingletonLock'), { force: true });
   545→    const timeZone = browserTimeZone(state.saved.timeZone);
   546→    const locale = browserLocale(state.saved.locale);
   547→    const result = await drainInterruptedCommand(state, action, () =>
   548→      run(
   549→        executable,
   550→        [
   551→          '--config',
   552→          config,
   553→          '--namespace',
   554→          namespace,
   555→          '--session',
   556→          browserSessionId(state.id),
   557→          '--profile',
   558→          profile,
   559→          ...(state.browserExecutable ? ['--executable-path', state.browserExecutable] : []),
   560→          ...(headed ? ['--headed'] : []),
   561→          '--no-webmcp',
   562→          '--init-script',
   563→          initScript,
   564→          '--json',
   565→          '--proxy',
   566→          proxyUrl,
   567→          '--proxy-bypass',
   568→          '<-loopback>',
   569→          '--args',
   570→          '--disable-quic,--force-webrtc-ip-handling-policy=disable_non_proxied_udp,--disable-background-networking,--disable-crash-reporter,--disable-breakpad,--disable-gpu-rasterization,--enable-unsafe-swiftshader,--disable-blink-features=AutomationControlled,--disable-features=BackForwardCache' +
   571→            `,--disk-cache-dir=${cachePath(state)},--disk-cache-size=67108864` +
   572→            (locale ? `,--accept-lang=${locale}` : '') +
   573→            (sandbox ? '' : ',--no-sandbox') +
   574→            (headed && platform === 'darwin' ? ',--start-minimized' : '') +
   575→            (state.saved.restoreSession ? ',--restore-last-session' : ''),
   576→          '--download-path',
   577→          resolve(runtimeRoot, 'downloads', state.id),
   578→          '--idle-timeout',
   579→          '86400s',
   580→          ...(action === 'close' ? ['close'] : ['open', String(request.url)]),
   581→        ],
   582→        {
   583→          cwd: runtimeRoot,
   584→          env: {
   585→            ...serviceEnvironment(),
   586→            ...(timeZone && { TZ: timeZone }),
   587→          },
   588→          signal: options.signal,
   589→          timeoutMs: options.timeoutMs ?? daemonDeadline(request) + 30_000,
   590→        },
   591→      ),
   592→    );
   593→    let response: { success?: boolean; data?: Record<string, unknown>; error?: unknown };
   594→    try {
   595→      response = JSON.parse(result.stdout.trim());
   596→    } catch {
   597→      throw new Error(
   598→        'The workspace browser could not respond. Restart its container and try again.',
   599→      );
   600→    }
   601→    if (action === 'navigate') {
   602→      await drainInterruptedCommand(state, 'connect', () => daemonSocket(state, options.signal));
   603→      state.recovered = false;
   604→      state.pageSize = undefined;
   605→      if (viewport)
   606→        await command(
   607→          state,
   608→          { action: 'viewport', width: viewport.width, height: viewport.height },
   609→          { signal: options.signal },
   610→        );
   611→    }
   612→    if (
   613→      action === 'navigate' &&
   614→      response.success === false &&
   615→      typeof response.error === 'string' &&
   616→      /^Navigation failed: net::ERR_[A-Z_]+/.test(response.error)
   617→    )
   618→      throw new BrowserActionError('browser_navigation_failed');
   619→    if (
   620→      action === 'navigate' &&
   621→      response.success === false &&
   622→      typeof response.error === 'string' &&
   623→      response.error.startsWith('Operation timed out.')
   624→    )
   625→      throw new DaemonCommandError();
   626→    if (result.exitCode || response.success === false)
   627→      throw new DaemonCommandError(typeof response.error === 'string' ? response.error : undefined);
   628→    return response.data ?? {};
   629→  }

   672→  async function currentUrl(state: BrowserState, signal?: AbortSignal): Promise<string> {
   673→    if (await pendingDialog(state, signal)) throw new BrowserActionError('browser_dialog_open');
   674→    const data = await observationCommand(state, { action: 'url' }, { signal });
   675→    const url = String(data.url ?? data.value ?? '');
   676→    const storedUrl = new URL(url);
   677→    if (!['http:', 'https:'].includes(storedUrl.protocol)) return url;
   678→    browserUrl(url);
   679→    storedUrl.search = '';
   680→    storedUrl.hash = '';
   681→    if (redactBrowserText(storedUrl.pathname, state.secrets) !== storedUrl.pathname)
   682→      storedUrl.pathname = '/';
   683→    state.saved.url = storedUrl.href;
   684→    return url;
   685→  }

   698→  async function recoverBrowser(
   699→    id: string,
   700→    allowSensitive = false,
   701→  ): Promise<BrowserState | undefined> {
   702→    threadIdentity(id);
   703→    const existing = browsers.get(id);
   704→    if (existing) return existing;
   705→    if (disposed) return undefined;
   706→    const saved = await screens.load(id);
   707→    if (
   708→      !saved ||
   709→      (!saved.active && !saved.navigationFailed) ||
   710→      (saved.sensitive && !allowSensitive)
   711→    )
   712→      return undefined;
   713→    try {
   714→      const result = await run(
   715→        executable,
   716→        ['--namespace', namespace, '--session', browserSessionId(id), 'session', 'info', '--json'],
   717→        {
   718→          env: serviceEnvironment(),
   719→          timeoutMs: 5000,
   720→        },
   721→      );
   722→      const response = JSON.parse(result.stdout);
   723→      const info = response.data;
   724→      if (
   725→        result.exitCode ||
   726→        !response.success ||
   727→        !info?.active ||
   728→        info.namespace !== namespace ||
   729→        info.session !== browserSessionId(id) ||
   730→        typeof info.socketDir !== 'string'
   731→      )
   732→        return undefined;
   733→      const state = stateFor(id);
   734→      state.saved = saved;
   735→      state.socket = resolve(info.socketDir, `${browserSessionId(id)}.sock`);
   736→      state.recovered = true;
   737→      state.started = !saved.navigationFailed;
   738→      return state;
   739→    } catch {
   740→      return undefined;
   741→    }
   742→  }

   744→  async function daemonSocket(state: BrowserState, signal?: AbortSignal): Promise<string> {
   745→    if (state.socket) return state.socket;
   746→    const result = await run(
   747→      executable,
   748→      [
   749→        '--namespace',
   750→        namespace,
   751→        '--session',
   752→        browserSessionId(state.id),
   753→        'session',
   754→        'info',
   755→        '--json',
   756→      ],
   757→      {
   758→        env: serviceEnvironment(),
   759→        timeoutMs: 5000,
   760→        signal,
   761→      },
   762→    );
   763→    const response = JSON.parse(result.stdout);
   764→    const info = response.data;
   765→    if (
   766→      result.exitCode ||
   767→      !response.success ||
   768→      !info?.active ||
   769→      info.namespace !== namespace ||
   770→      info.session !== browserSessionId(state.id) ||
   771→      typeof info.socketDir !== 'string'
   772→    )
   773→      throw new Error('The browser session is no longer available.');
   774→    state.socket = resolve(info.socketDir, `${browserSessionId(state.id)}.sock`);
   775→    return state.socket;
   776→  }

   778→  async function observe(
   779→    state: BrowserState,
   780→    signal?: AbortSignal,
   781→    input?: BrowserAction,
   782→  ): Promise<BrowserObservation> {
   783→    if (state.injecting || (await privateDocumentOpen(state, signal)))
   784→      throw new BrowserLockedError();
   785→    let url: string;
   786→    try {
   787→      url = await currentUrl(state, signal);
   788→    } catch (error) {
   789→      const dialog =
   790→        error instanceof BrowserActionError &&
   791→        error.code === 'browser_dialog_open' &&
   792→        (await pendingDialog(state, signal));
   793→      if (dialog) return { text: dialog, url: state.saved.url };
   794→      throw error;
   795→    }
   796→    if (url === 'about:blank')
   797→      return {
   798→        text: 'The page opened a new tab that has not loaded. Take a snapshot again.',
   799→        url,
   800→      };
   801→    const scope =
   802→      input?.action === 'snapshot' && input.selector ? selector(input.selector) : undefined;
   803→    const offset = input?.action === 'snapshot' ? (input.offset ?? 0) : 0;
   804→    if (!Number.isInteger(offset) || offset < 0)
   805→      throw new Error('Choose a non-negative page offset.');
   806→    const snapshot = async () => {
   807→      const devtools = state.secrets.size ? await inspector(state) : undefined;
   808→      try {
   809→        await devtools?.mask(true, [], false, true);
   810→        return await observationCommand(
   811→          state,
   812→          { action: 'snapshot', urls: true, ...(scope ? { selector: scope } : {}) },
   813→          { signal },
   814→        );
   815→      } finally {
   816→        await devtools?.mask(false, [], false).catch(() => {});
   817→        devtools?.close();
   818→      }
   819→    };
   820→    const data = await snapshot();
   821→    if (await privateDocumentOpen(state, signal)) throw new BrowserLockedError();
   822→    const incoming =
   823→      data.refs && typeof data.refs === 'object' ? (data.refs as Record<string, unknown>) : {};
   824→    if (!scope) {
   825→      state.refs.clear();
   826→      for (const ref of state.refVersions.keys())
   827→        if (!(ref in incoming)) state.refVersions.delete(ref);
   828→    }
   829→    for (const ref of Array.isArray(data.removedRefs) ? data.removedRefs : []) {
   830→      state.refs.delete(String(ref));
   831→      state.refVersions.delete(String(ref));
   832→    }
   833→    for (const [ref, control] of Object.entries(incoming))
   834→      if (control && typeof control === 'object') {
   835→        if (!state.refVersions.has(ref)) state.refVersions.set(ref, ++refSequence);
   836→        state.refs.set(ref, {
   837→          role: String((control as { role?: unknown }).role ?? ''),
   838→          name: redactBrowserText(
   839→            String((control as { name?: unknown }).name ?? ''),
   840→            state.secrets,
   841→          ).slice(0, 300),
   842→        });
   843→      }
   844→    const page = redactBrowserText(String(data.snapshot ?? ''), state.secrets);
   845→    const end = Math.min(page.length, offset + snapshotChars);
   846→    return {
   847→      text: `Page (${offset}-${end} of ${page.length} characters${end < page.length ? `; snapshot with offset ${end} continues` : ''}):\n${page.slice(offset, end)}`,
   848→      url: redactBrowserUrl(url, state.secrets),
   849→    };
   850→  }

   852→  async function movePointer(
   853→    state: BrowserState,
   854→    gesture: NonNullable<ReturnType<typeof visualGesture>>,
   855→    signal?: AbortSignal,
   856→  ) {
   857→    const [width, height] = z
   858→      .tuple([z.number().positive(), z.number().positive()])
   859→      .parse(await evaluate(state, '[innerWidth, innerHeight]', signal));
   860→    const move = (point: { x: number; y: number }) =>
   861→      command(
   862→        state,
   863→        {
   864→          action: 'mousemove',
   865→          x: Math.min(Math.ceil(width) - 1, Math.round(point.x * width)),
   866→          y: Math.min(Math.ceil(height) - 1, Math.round(point.y * height)),
   867→        },
   868→        { signal },
   869→      );
   870→    await move(gesture.points[0]);
   871→    try {
   872→      await command(state, { action: 'mousedown', button: 'left' }, { signal });
   873→      if (gesture.points.length === 1 && gesture.durationMs)
   874→        await delay(gesture.durationMs, undefined, { signal });
   875→      for (const point of gesture.points.slice(1)) {
   876→        if (gesture.durationMs)
   877→          await delay(gesture.durationMs / (gesture.points.length - 1), undefined, { signal });
   878→        await move(point);
   879→      }
   880→    } finally {
   881→      if (state.started && !state.closed && !state.closing) {
   882→        try {
   883→          await command(state, { action: 'mouseup', button: 'left' }, { timeoutMs: 5000 });
   884→        } catch (error) {
   885→          if (!state.closing) await closeInterruptedBrowser(state);
   886→          throw error;
   887→        }
   888→      }
   889→    }
   890→  }

   892→  async function browserAction(
   893→    id: string,
   894→    input: BrowserAction,
   895→    signal?: AbortSignal,
   896→  ): Promise<BrowserObservation> {
   897→    if (disposed) throw new Error('The browser service has stopped.');
   898→    const gesture = visualGesture(input);
   899→    if (input.waitMs !== undefined && (!Number.isInteger(input.waitMs) || input.waitMs < 0))
   900→      throw new Error('Choose a non-negative wait.');
   901→    const savedProfile = input.action === 'open' ? await screens.load(id) : null;
   902→    if (input.action === 'open' && !browsers.has(id)) await recoverBrowser(id, true);
   903→    const existing = input.action === 'open' ? browsers.get(id) : undefined;
   904→    if (
   905→      existing &&
   906→      (existing.recovered ||
   907→        existing.closed ||
   908→        existing.closing ||
   909→        existing.saved.navigationFailed ||
   910→        existing.privateDocument)
   911→    )
   912→      await closeBrowser(id);
   913→    const state = stateFor(id);
   914→    if (input.action === 'open' && savedProfile?.profileId && !state.saved.profileId) {
   915→      state.saved.profileId = savedProfile.profileId;
   916→      state.saved.restoreSession = savedProfile.restoreSession;
   917→      state.saved.sensitive = savedProfile.sensitive;
   918→    }
   919→    if (input.action !== 'open' && !state.started) {
   920→      const saved = await screens.load(id);
   921→      if (saved?.navigationFailed) state.saved = saved;
   922→    }
   923→    if (state.injecting) throw new BrowserLockedError();
   924→    return serialize(state, async () => {
   925→      if (state.injecting) throw new BrowserLockedError();
   926→      signal?.throwIfAborted();
   927→      if (input.action === 'open') {
   928→        const url = browserUrl(input.url ?? '');
   929→        const previous = state.saved.restoreSession ? state.saved : await screens.load(id);
   930→        if (previous?.restoreSession) {
   931→          state.saved.restoreSession = true;
   932→          state.saved.sensitive = true;
   933→        }
   934→        if (state.closing || disposed) throw new BrowserLockedError();
   935→        signal?.throwIfAborted();
   936→        const resumeUrl = new URL(url);
   937→        resumeUrl.search = '';
   938→        resumeUrl.hash = '';
   939→        if (redactBrowserText(resumeUrl.pathname, state.secrets) !== resumeUrl.pathname)
   940→          resumeUrl.pathname = '/';
   941→        state.saved.resumeUrl = resumeUrl.href;
   942→        await reserveBrowser(state, signal);
   943→        if (!state.socket) {
   944→          state.saved.timeZone = browserTimeZone(input.timeZone);
   945→          state.saved.locale = browserLocale(input.locale);
   946→        }
   947→        try {
   948→          if (!state.socket && viewport)
   949→            await command(state, { action: 'navigate', url: 'about:blank' }, { signal });
   950→          await command(state, { action: 'navigate', url: url.href }, { signal });
   951→          state.saved.navigationFailed = false;
   952→        } catch (error) {
   953→          if (
   954→            (error instanceof DaemonCommandError ||
   955→              (error instanceof BrowserActionError &&
   956→                error.code === 'browser_navigation_failed')) &&
   957→            !signal?.aborted
   958→          ) {
   959→            try {
   960→              const loaded = new URL(await currentUrl(state, signal));
   961→              if (['http:', 'https:'].includes(loaded.protocol)) {
   962→                const observation = await observe(state, signal);
   963→                if (observation.text.slice(observation.text.indexOf('\n') + 1).trim()) {
   964→                  state.saved.navigationFailed = false;
   965→                  return {
   966→                    ...observation,
   967→                    text: `Opening ${redactBrowserUrl(url.href, state.secrets)} did not finish loading. This is the page the browser shows now.\n${observation.text}`,
   968→                  };
   969→                }
   970→              }
   971→            } catch {}
   972→          }
   973→          if (!signal?.aborted && !state.closing) {
   974→            state.started = false;
   975→            state.saved.navigationFailed = true;
   976→            const destination = new URL(url);
   977→            destination.search = '';
   978→            destination.hash = '';
   979→            if (redactBrowserText(destination.pathname, state.secrets) !== destination.pathname)
   980→              destination.pathname = '/';
   981→            state.saved.url = destination.href;
   982→            await saveState(state);
   983→          }
   984→          throw error;
   985→        }
   986→      } else {
   987→        if (state.saved.navigationFailed) {
   988→          if (input.action !== 'snapshot')
   989→            throw new BrowserActionError('browser_navigation_failed');
   990→          return {
   991→            text: new BrowserActionError('browser_navigation_failed').message,
   992→            url: state.saved.url,
   993→          };
   994→        }
   995→        if (!state.started) throw new BrowserActionError('browser_not_open');
   996→        if (input.action !== 'reload' && (await privateDocumentOpen(state, signal)))
   997→          throw new BrowserActionError('browser_card_paused');
   998→        await onTarget(state, input, signal);
   999→        if (input.action === 'tab_list') {
  1000→          const tabs = await browserTabs(state, signal);
  1001→          return {
  1002→            text: tabs
  1003→              .map(
  1004→                (tab) =>
  1005→                  `${tab.active ? 'Active' : 'Open'} ${tab.tabId}: ${redactBrowserUrl(tab.url, state.secrets)}`,
  1006→              )
  1007→              .join('\n'),
  1008→            url: redactBrowserUrl(await currentUrl(state, signal), state.secrets),
  1009→          };
  1010→        }
  1011→        if (input.action === 'tab_switch') {
  1012→          await command(state, { action: 'tab_switch', tabId: input.tabId }, { signal });
  1013→        }
  1014→        if (input.action === 'click') {
  1015→          if (gesture) await movePointer(state, gesture, signal);
  1016→          else
  1017→            await command(
  1018→              state,
  1019→              { action: 'click', selector: selector(input.selector) },
  1020→              { signal },
  1021→            );
  1022→        }
  1023→        if (input.action === 'fill' || input.action === 'type') {
  1024→          const target = selector(input.selector);
  1025→          const text = input.text ?? '';
  1026→          await command(
  1027→            state,
  1028→            input.action === 'fill'
  1029→              ? { action: 'fill', selector: target, value: text }
  1030→              : { action: 'type', selector: target, text },
  1031→            { signal },
  1032→          );
  1033→        }
  1034→        if (input.action === 'scroll')
  1035→          await command(
  1036→            state,
  1037→            {
  1038→              action: 'scroll',
  1039→              direction: input.direction ?? 'down',
  1040→              amount: input.amount ?? 600,
  1041→              ...(input.selector ? { selector: selector(input.selector) } : {}),
  1042→            },
  1043→            { signal },
  1044→          );
  1045→        if (input.action === 'press') {
  1046→          if (!input.key) throw new Error('Choose a key to press.');
  1047→          await command(state, { action: 'press', key: input.key }, { signal });
  1048→        }
  1049→        if (input.action === 'accept_dialog' || input.action === 'dismiss_dialog') {
  1050→          const text = input.action === 'accept_dialog' ? input.text : undefined;
  1051→          await command(
  1052→            state,
  1053→            {
  1054→              action: 'dialog',
  1055→              response: input.action === 'accept_dialog' ? 'accept' : 'dismiss',
  1056→              ...(text ? { promptText: text } : {}),
  1057→            },
  1058→            { signal },
  1059→          );
  1060→        }
  1061→        if (input.action === 'wait') {
  1062→          if (
  1063→            [input.text, input.url, input.load].filter(Boolean).length !== 1 ||
  1064→            (input.load && !['load', 'domcontentloaded', 'networkidle'].includes(input.load))
  1065→          )
  1066→            throw new Error('Wait for one of text, url or load.');
  1067→          try {
  1068→            await command(
  1069→              state,
  1070→              {
  1071→                ...(input.text
  1072→                  ? { action: 'wait', text: input.text }
  1073→                  : input.url
  1074→                    ? { action: 'waitforurl', url: input.url }
  1075→                    : { action: 'waitforloadstate', state: input.load }),
  1076→                ...(input.waitMs === undefined ? {} : { timeout: input.waitMs }),
  1077→              },
  1078→              { signal },
  1079→            );
  1080→          } catch (error) {
  1081→            if (!(error instanceof DaemonCommandError)) throw error;
  1082→            const observation = await observe(state, signal);
  1083→            return {
  1084→              ...observation,
  1085→              text: `The wait condition was not met.\n${observation.text}`,
  1086→            };
  1087→          }
  1088→        }
  1089→        if (input.action === 'select') {
  1090→          const value = input.text ?? '';
  1091→          if (!value) throw new Error('Choose a visible option.');
  1092→          await command(
  1093→            state,
  1094→            { action: 'select', selector: selector(input.selector), value },
  1095→            { signal },
  1096→          );
  1097→        }
  1098→        if (['check', 'uncheck', 'hover'].includes(input.action))
  1099→          await command(
  1100→            state,
  1101→            { action: input.action, selector: selector(input.selector) },
  1102→            { signal },
  1103→          );
  1104→        if (input.action === 'drag') {
  1105→          if (gesture) await movePointer(state, gesture, signal);
  1106→          else
  1107→            await command(
  1108→              state,
  1109→              {
  1110→                action: 'drag',
  1111→                source: selector(input.selector),
  1112→                target: selector(input.targetSelector),
  1113→              },
  1114→              { signal },
  1115→            );
  1116→        }
  1117→        if (['back', 'forward', 'reload'].includes(input.action)) {
  1118→          const devtools = await inspector(state, signal).catch(() => undefined);
  1119→          try {
  1120→            const before = await devtools?.document().catch(() => undefined);
  1121→            const entry =
  1122→              input.action === 'reload'
  1123→                ? await devtools?.navigation().catch(() => undefined)
  1124→                : undefined;
  1125→            const native =
  1126→              input.action !== 'reload' || (!!entry?.type && entry.type !== 'form_submit');
  1127→            const target = native
  1128→              ? undefined
  1129→              : browserUrl(entry?.url ?? (await currentUrl(state, signal))).href;
  1130→            const failure = await command(
  1131→              state,
  1132→              target ? { action: 'navigate', url: target } : { action: input.action },
  1133→              { signal },

  1136→              (error: unknown) => error,
  1137→            );

  1145→            }

  1147→            devtools?.close();

  1159→              if (!current || current === state.sealedDocument)
  1160→                throw new BrowserActionError('browser_script_paused');
  1161→              delete state.sealedDocument;
  1162→            }

  1164→              throw new BrowserActionError('browser_script_paused');

  1166→            if (!read) {
  1167→              if (input.readOnly) throw new BrowserActionError('browser_script_writes');
  1168→              await devtools.scripted(true);
  1169→              await saveState(state);
  1170→            }
  1171→          } finally {
  1172→            devtools.close();
  1173→          }
  1174→          const value = read ? read.value : await evaluate(state, script, signal);
  1175→          if (!read) {
  1176→            await markScripted(state);
  1177→            await observe(state, signal).catch(() => state.refs.clear());
  1178→          }
  1179→          const result = redactBrowserText(JSON.stringify(value ?? null) ?? 'null', state.secrets);
  1180→          return {
  1181→            text: `Script result (JSON):\n${result}`,
  1182→            url: redactBrowserUrl(url, state.secrets),
  1183→          };
  1184→        }

  1186→      if (input.waitMs && input.action !== 'wait') await delay(input.waitMs, undefined, { signal });

  1190→        throw new BrowserActionError(
  1191→          'browser_action_failed',
  1192→          error.detail ? redactBrowserText(error.detail, state.secrets) : undefined,
  1193→        );
  1194→      throw error;
  1195→    });
  1196→  }

  1355→  async function privateEntry<T>(
  1356→    state: BrowserState,
  1357→    submitSelector: string | undefined,
  1358→    operation: (devtools: Awaited<ReturnType<typeof openDevtools>>, document: string) => Promise<T>,
  1359→    submitOrigins: string[] = [],
  1360→  ): Promise<T> {
  1361→    if (state.injecting) throw new BrowserLockedError();
  1362→    state.injecting = true;
  1363→    try {
  1364→      return await serialize(state, async () => {
  1365→        const devtools = await inspector(state);
  1366→        try {
  1367→          const document = await checkout(state, devtools);
  1368→          state.privateDocument = document;
  1369→          state.saved.sensitive = true;
  1370→          await saveState(state);
  1371→          const result = await operation(devtools, document);
  1372→          if (submitSelector && (await devtools.document()) !== document)
  1373→            throw new BrowserActionError('browser_card_changed');
  1374→          if ((await devtools.document()) === document) {
  1375→            if (submitSelector) {
  1376→              const field = await focusField(state, devtools, selector(submitSelector));
  1377→              const origin = new URL(await currentUrl(state)).origin;
  1378→              if (
  1379→                (field.origin !== origin && !submitOrigins.includes(field.origin)) ||
  1380→                (await devtools.document()) !== document
  1381→              )
  1382→                throw new Error('The submit control is not on this page.');
  1383→              await command(state, {
  1384→                action: 'scrollintoview',
  1385→                selector: selector(submitSelector),
  1386→              });
  1387→              await command(state, { action: 'click', selector: selector(submitSelector) });
  1388→            }
  1389→          }
  1390→          state.privateDocument = undefined;
  1391→          if (state.secrets.size) state.sealedDocument = document;
  1392→          return result;
  1393→        } finally {
  1394→          devtools.close();
  1395→        }
  1396→      });
  1397→    } finally {
  1398→      state.injecting = false;
  1399→    }
  1400→  }

  1562→  async function fillCard(input: CardEntry): Promise<void> {
  1563→    const state = browsers.get(threadIdentity(input.threadId));
  1564→    if (!state?.started) throw new Error('The checkout page is no longer open.');
  1565→    if (state.injecting) throw new BrowserLockedError();
  1566→    const { card, fields } = input;
  1567→    if (fields.securityCode && !card.securityCode)
  1568→      throw new Error('The security code is needed for this card.');
  1569→    const month = String(card.expiryMonth).padStart(2, '0');
  1570→    const year = String(card.expiryYear);
  1571→    const monthName = (length: 'long' | 'short', locale = 'en') =>
  1572→      new Intl.DateTimeFormat(locale, { month: length, timeZone: 'UTC' }).format(
  1573→        new Date(Date.UTC(card.expiryYear, card.expiryMonth - 1, 1)),
  1574→      );
  1575→    const expiry = (field: CardExpiryField | undefined, defaults: string[]) => {
  1576→      if (!field || typeof field === 'string') return defaults;
  1577→      return [
  1578→        field.format.replace(/MMMM|MMM|MM|M|YYYY|YY/g, (token) => {
  1579→          if (token === 'M') return String(card.expiryMonth);
  1580→          if (token === 'MM') return month;
  1581→          if (token === 'YYYY') return year;
  1582→          if (token === 'YY') return year.slice(-2);
  1583→          return monthName(token === 'MMM' ? 'short' : 'long', field.locale);
  1584→        }),
  1585→      ];
  1586→    };
  1587→    const plan: Array<{ field: keyof CardFields; values: string[] }> = [
  1588→      { field: 'number', values: [card.cardNumber] },
  1589→      { field: 'expiry', values: expiry(fields.expiry, [`${month}/${year.slice(-2)}`]) },
  1590→      {
  1591→        field: 'expiryMonth',
  1592→        values: expiry(fields.expiryMonth, [
  1593→          month,
  1594→          String(card.expiryMonth),
  1595→          monthName('long'),
  1596→          monthName('short'),
  1597→        ]),
  1598→      },
  1599→      { field: 'expiryYear', values: expiry(fields.expiryYear, [year, year.slice(-2)]) },
  1600→      { field: 'securityCode', values: card.securityCode ? [card.securityCode] : [] },
  1601→      { field: 'name', values: [card.cardholder] },
  1602→    ];
  1603→    const steps = plan.filter((step) => fields[step.field] && step.values.length);
  1604→    for (const step of steps)
  1605→      if (!input.destinations[step.field])
  1606→        throw new Error('The approved card destinations are missing.');
  1607→    await privateEntry(
  1608→      state,
  1609→      input.submitSelector,
  1610→      async (devtools, document) => {
  1611→        if (document !== input.document) throw new BrowserActionError('browser_card_changed');
  1612→        for (const step of steps) {
  1613→          const privateValue = step.field === 'number' || step.field === 'securityCode';
  1614→          if (privateValue) for (const value of step.values) state.secrets.add(value);
  1615→          const selection = fields[step.field]!;
  1616→          const field = await focusField(

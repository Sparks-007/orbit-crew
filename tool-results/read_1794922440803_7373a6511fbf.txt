   144→// The rail label rides along with each id, because the user names a tab by what they read on
   145→// screen ("Apps", "Capabilities"), not by its id.
   146→function describeTabs<T extends string>(tabs: readonly T[], labels: Record<T, string>, hints: Record<T, string>): string {
   147→    return tabs.map(tab => `\`${tab}\` ("${labels[tab]}": ${hints[tab]})`).join(', ')
   148→}

   168→// What a plan is allowed to hold, narrowed to the kinds this conversation can really land.
   169→//
   170→// The narrowing is the whole point: a team is designed, not picked, so a funnel that never met a
   171→// ready-made one reads a schema where an agent is the only shape a plan item has. A template it
   172→// cannot write down is a template the build phase never re-arms install_template for.
   173→function planItemsSchema(kinds: readonly PlanItemKind[], pinned: readonly string[] | null): object {
   174→    const ready = kinds.includes('template')
   175→    const shared = kinds.includes('share')
   176→    const box = pinned ? { type: 'string', enum: [...pinned], description: 'The Box its work belongs to, one of the Boxes already on their board.' } : { type: 'string', description: 'The Box this one lands in, named exactly as it appears in boxes.' }
   177→
   178→    return {
   179→        type: 'array',
   180→        maxItems: MAX_PLAN_ITEMS,
   181→        items: {
   182→            type: 'object',
   183→            properties: {
   184→                kind: { type: 'string', enum: [...kinds] },
   185→                ...(ready ? { slug: { type: 'string', description: 'For kind template: the library slug.' }, title: { type: 'string', description: 'For kind template: its title, for the record.' } } : {}),
   186→                ...(shared ? { token: { type: 'string', description: 'For kind share: the token of the link they arrived on.' } } : {}),
   187→                name: { type: 'string', description: 'For kind agent: what it will be called.' },
   188→                role: { type: 'string', description: 'For kind agent: what it hands over, in one plain line of twelve words at most that starts with a verb: "Writes the case study for each finished project". No metaphor, no dash.' },
   189→                need: { type: 'string', description: 'For kind agent: the pain from the interview this coworker goes after, in their words. Leave it out when the interview dug out none for its job.' },
   190→                when: { type: 'string', description: 'For kind agent: what sets it off or how often it works, in a few words: "Every Monday", "When a brief comes in".' },
   191→                beats: {
   192→                    type: 'object',
   193→                    description: 'For kind agent: one ordinary run of it at their company, played in a loop on the card. Three lines of five words at most, concrete enough to be theirs, and never naming an app.',
   194→                    properties: {
   195→                        trigger: { type: 'string', description: 'What just set it off: "New email: where is my order?", "Monday 8am, weekly run".' },
   196→                        working: { type: 'string', description: 'What it is seen doing: "Reading order #4812", "Scoring 42 new projects".' },
   197→                        delivered: { type: 'string', description: 'What it hands over: "Reply drafted for you", "6 leads worth a quote".' },
   198→                    },
   199→                    required: ['trigger', 'working', 'delivered'],
   200→                },
   201→                gets: {
   202→                    type: 'array',
   203→                    maxItems: 3,
   204→                    items: { type: 'string' },
   205→                    description: 'For kind agent: what they get from it this week, three plain lines of ten words at most, each a thing it hands over: "A drafted reply on every order question". Never an app, never a promise it cannot keep alone.',
   206→                },
   207→                box,
   208→            },
   209→            required: pinned ? ['kind', 'box'] : ['kind'],
   210→        },
   211→    }

    97→export function compactHistory(history: RuntimeContext['history']) {
    98→  let start = history.length;
    99→  let characters = 0;
   100→  let exchanges = 0;
   101→  for (let index = history.length - 1; index >= 0; index--) {
   102→    if (history[index].role !== 'user') continue;
   103→    characters += JSON.stringify(history.slice(index, start)).length;
   104→    if (characters > 4000) break;
   105→    start = index;
   106→    if (++exchanges === 2) break;
   107→  }
   108→  return history.slice(start);
   109→}

   166→function unanswered(context: RuntimeContext) {
   167→  const lastUser = context.history.findLastIndex((item) => item.role === 'user');
   168→  if (lastUser < 0) return '';
   169→  const sent = new Set(
   170→    context.history
   171→      .slice(lastUser + 1)
   172→      .flatMap((item, index) =>
   173→        item.role === 'assistant' ? [item.turnId ?? `message-${index}`] : [],
   174→      ),
   175→  ).size;
   176→  if (!sent) return '';
   177→  const since = context.history[lastUser].createdAt;
   178→  const hours = since ? Math.floor((Date.now() - new Date(since).getTime()) / 3_600_000) : null;
   179→  const ago =
   180→    hours === null
   181→      ? ''
   182→      : ` (their last message was ${hours < 1 ? 'less than an hour' : hours === 1 ? '1 hour' : `${hours} hours`} ago)`;

   189→function memoryForPrompt(brief: NonNullable<RuntimeContext['memoryBrief']>, budget: number) {
   190→  if (JSON.stringify(brief).length <= budget) return brief;
   191→  const result = {
   192→    mode: brief.mode,
   193→    shortened: true,
   194→    facts: [] as object[],
   195→    messages: [] as object[],
   196→    searches: [] as object[],
   197→  };
   198→  const keep = (target: object[], item: object) => {
   199→    target.push(item);
   200→    if (JSON.stringify(result).length > budget) target.pop();
   201→  };
   202→  for (const note of brief.facts) {
   203→    keep(result.facts, {
   204→      path: note.path,
   205→      text: note.text,
   206→      status: note.status,
   207→      fact: note.fact,
   208→      sources: note.sources.map(({ id, at }) => ({ id, at })),
   209→    });
   210→  }
   211→  for (const message of brief.messages) {
   212→    keep(result.messages, {
   213→      ...message,
   214→      text: message.text.slice(0, 400),
   215→      shortened: message.shortened || message.text.length > 400,
   216→    });
   217→  }
   218→  for (const search of brief.searches) keep(result.searches, search);
   219→  return result;
   220→}

   363→export function observerInstruction(
   364→  context: RuntimeContext,
   365→  resumed: boolean,
   366→  time: string,
   367→  sourceReads?: string,
   368→) {
   369→  const parts = requestParts(context, resumed, new Set());
   370→  const prefix = (value: object) => `${parts.preamble}${JSON.stringify(value).slice(0, -1)},`;
   371→  const { recentConversation, currentInstruction, ...tail } = parts.volatile;
   372→  const request = {
   373→    ...parts.stable,
   374→    recentConversation,
   375→    ...tail,
   376→    ...(sourceReads ? { sourceReads } : {}),
   377→    currentInstruction,
   378→  };
   379→  return {
   380→    stable: [prefix(parts.stable), prefix({ ...parts.stable, recentConversation })],
   381→    text: `${parts.preamble}${JSON.stringify(request)}\n${context.credentialRequests?.length ? credentialStatusGuide : credentialStatusCaution}\n${time}`,
   382→  };
   383→}

   385→function workspaceAttachments(
   386→  attachments: RuntimeContext['attachments'],
   387→  omittedImages: Set<string>,
   388→) {
   389→  return attachments.map((item) => ({
   390→    name: basename(item.path),
   391→    mimeType: item.mimeType,
   392→    ...(item.source ? { source: item.source } : {}),
   393→    ...(item.workspaceFile ? { workspaceFile: item.workspaceFile } : {}),
   394→    ...(omittedImages.has(item.path) ? { imagePreview: 'omitted' } : {}),
   395→  }));
   396→}

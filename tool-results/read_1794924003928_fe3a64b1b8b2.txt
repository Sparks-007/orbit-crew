   114→export function endpointAddress(address: string) {
   115→  let url: URL;
   116→  try {
   117→    url = new URL(
   118→      /^[a-z][a-z\d+.-]*:/i.test(address) && !/^[^/:]+:\d+(\/|$)/.test(address)
   119→        ? address
   120→        : `http://${address}`,
   121→    );
   122→  } catch {
   123→    return undefined;
   124→  }
   125→  if (
   126→    !['http:', 'https:'].includes(url.protocol) ||
   127→    url.username ||
   128→    url.password ||
   129→    /[?#]/.test(url.href)
   130→  )
   131→    return undefined;
   132→  return url.href.replace(/\/+$/, '');
   133→}

   135→function azureAddress(address: string) {
   136→  const value = address.trim();
   137→  const url = endpointAddress(/^[a-z][a-z\d+.-]*:\/\//i.test(value) ? value : `https://${value}`);
   138→  if (!url) return undefined;
   139→  const { protocol, hostname, origin, pathname } = new URL(url);
   140→  if (protocol !== 'https:' && !['localhost', '127.0.0.1', '[::1]'].includes(hostname))
   141→    return undefined;
   142→  return pathname === '/' || pathname === '/openai' ? `${origin}/openai/v1` : url;
   143→}

   145→export function modelKeyCredential(
   146→  provider: ModelProvider,
   147→  key: string,
   148→  settings: {
   149→    region?: string;
   150→    url?: string;
   151→    model?: string;
   152→    deployment?: string;
   153→    accountId?: string;
   154→    gatewayId?: string;
   155→    workspaceId?: string;
   156→  },
   157→): ModelCredential {
   158→  if (provider === 'openai') {
   159→    const region = settings.region || 'global';
   160→    if (!Object.hasOwn(openAIRegions, region))
   161→      throw new ConnectionError('Choose an OpenAI API region.');
   162→    return {
   163→      type: 'api_key',
   164→      key,
   165→      ...(region !== 'global' ? { region: region as keyof typeof openAIRegions } : {}),
   166→    };
   167→  }
   168→  if (provider === 'anthropic') {
   169→    const workspaceId = settings.workspaceId?.trim();
   170→    if (workspaceId && (workspaceId.length > 100 || !/^[\x21-\x7e]+$/.test(workspaceId)))
   171→      throw new ConnectionError('Enter a Workspace ID from the Claude Console.');
   172→    return { type: 'api_key', key, ...(workspaceId ? { workspaceId } : {}) };
   173→  }
   174→  if (provider === 'amazon-bedrock') {
   175→    if (!settings.region || !bedrockRegionPattern.test(settings.region))
   176→      throw new ConnectionError('Enter an AWS Region, such as us-east-1.');
   177→    return { type: 'bedrock', key, region: settings.region };
   178→  }
   179→  if (provider === 'cloudflare-workers-ai' || provider === 'cloudflare-ai-gateway') {
   180→    if (!settings.accountId || !cloudflareAccountPattern.test(settings.accountId))
   181→      throw new ConnectionError('Enter your 32-character Cloudflare account ID.');
   182→    if (
   183→      provider === 'cloudflare-ai-gateway' &&
   184→      (!settings.gatewayId || !cloudflareGatewayPattern.test(settings.gatewayId))
   185→    )
   186→      throw new ConnectionError('Enter your Cloudflare AI Gateway ID.');
   187→    return {
   188→      type: 'cloudflare',
   189→      key,
   190→      accountId: settings.accountId.toLowerCase(),
   191→      ...(provider === 'cloudflare-ai-gateway' ? { gatewayId: settings.gatewayId } : {}),
   192→    };
   193→  }
   194→  if (provider !== 'azure') return { type: 'api_key', key };
   195→  const url = settings.url && azureAddress(settings.url);
   196→  if (!url)
   197→    throw new ConnectionError(
   198→      'Enter your Azure OpenAI endpoint, such as https://your-resource.openai.azure.com.',
   199→    );
   200→  const deployment = settings.deployment || settings.model;
   201→  if (!settings.model || !deployment || !azureDeploymentPattern.test(deployment))
   202→    throw new ConnectionError('Choose the model and enter its deployment name.');
   203→  return { type: 'azure', url, key, model: settings.model, deployment };
   204→}

   207→  row: ModelRecord,
   208→  provider: ModelProvider,
   209→  method: ModelMethod,

   216→  return JSON.stringify(saved) === JSON.stringify(value);

   232→function textCompanion(row: Pick<ModelRecord, 'provider' | 'method' | 'models'>) {
   233→  const candidates = row.models.filter(

   429→  async function reload() {
   430→    const state = await store.read();
   431→    rows = state.connections.map(({ credential: saved, ...row }) => {
   432→      const value = row.managed ? options.bootstrap?.credentials[row.provider] : saved;
   433→      const failed = failures(row, credentialTag(value)).sort(
   434→        ([, a], [, b]) => b.retryAt - a.retryAt,
   435→      );
   436→      return {
   437→        ...row,
   438→        ...(value &&
   439→        typeof value === 'object' &&
   440→        'key' in value &&
   441→        typeof value.key === 'string' &&
   442→        value.key.length > 4
   443→          ? { keyEnding: value.key.slice(-4) }
   444→          : {}),
   445→        ...(failed.length
   446→          ? {
   447→              issue: { reason: failed[0]![1].reason },
   448→              issues: Object.fromEntries(failed.map(([id, item]) => [id, { reason: item.reason }])),
   449→              retry: Object.fromEntries(failed.map(([id, item]) => [id, item.retryAt])),
   450→            }
   451→          : {}),
   452→      };
   453→    });

   455→    routes = state.routes;
   456→    checkSelection(state);

   489→  function requestAccess(
   490→    row: ModelRecord,
   491→  ): Pick<ModelAccess, 'read' | 'prepareRequest' | 'reserve'> {
   492→    return {
   493→      read: () => read(row.id),
   494→      reserve,
   495→      async prepareRequest(modelId, automatic) {
   496→        const tag = credentialTag(await read(row.id));
   497→        const failure = new Map(
   498→          failures(
   499→            rows.find((item) => item.id === row.id),
   500→            tag,
   501→          ),
   502→        ).get(modelId);
   503→        if (automatic && failure && Date.now() < failure.retryAt)
   504→          throw new ConnectionError(
   505→            `${modelProviderNames[row.provider]} failed the last request to ${modelId} (${failure.reason}). Automatic requests to this model are paused until a request succeeds. Check this connection in ${appPages['/settings/models']} or ask again to make a new request.`,
   506→          );
   507→        return async (error) => {
   508→          if (!tag || closed) return;
   509→          const recorded = failures(
   510→            rows.find((item) => item.id === row.id),
   511→            tag,
   512→          );
   513→          if (!error && !recorded.some(([id]) => id === modelId)) return;
   514→          try {
   515→            await store.change((state) => {
   516→              const saved = state.connections.find((item) => item.id === row.id);
   517→              if (!saved || credentialTag(credential(saved)) !== tag) return;
   518→              const current = new Map(failures(saved, tag));
   519→              if (!error) {
   520→                if (current.delete(modelId))
   521→                  for (const item of current.values())
   522→                    item.retryAt = Math.min(item.retryAt, Date.now());
   523→                saved.requestFailures = Object.fromEntries(current);
   524→                return;
   525→              }
   526→              const previous = current.get(modelId);
   527→              const count =
   528→                previous && Date.now() < previous.retryAt
   529→                  ? previous.failures
   530→                  : (previous?.failures ?? 0) + 1;
   531→              saved.requestFailures = Object.fromEntries(current);
   532→              saved.requestFailures[modelId] = {
   533→                credentialTag: tag,
   534→                failures: count,
   535→                retryAt: Date.now() + failureWindow(count),
   536→                reason: redactSecretText(modelFailureReason(error))
   537→                  .replace(/\s*\[request phase: [^\]]*\]\s*$/, '')
   538→                  .replace(/\s+/g, ' ')
   539→                  .trim()
   540→                  .slice(0, 300),
   541→              };
   542→            });
   543→            await reload();
   544→          } catch (error) {
   545→            logFailure('model_failure_save_failed', error);
   546→          }
   547→        };
   548→      },
   549→    };
   550→  }
   551→  function apiKeyConnection(provider: ModelProvider) {
   552→    const candidates = rows.filter(
   553→      (row) =>

   579→  function classificationModel(otherConnections = false) {
   580→    const selected = selection && { ...selection };
   581→    const current = rows.find((row) => row.id === selected?.connectionId);
   582→    const main = current?.models.find((model) => model.choice.id === selected?.modelId);
   583→    const price = main && modelPricing(main);
   584→    if (
   585→      current?.provider === 'openrouter' &&
   586→      price &&
   587→      price !== 'variable' &&
   588→      price.input + price.output === 0
   589→    )
   590→      return undefined;
   591→    const candidates = [
   592→      current,
   593→      ...(otherConnections ? rows.filter((row) => row !== current) : []),
   594→    ];
   595→    for (const row of candidates) {
   596→      if (!row || row.status !== 'connected') continue;
   597→      const definition = row.models.find(
   598→        (model) => model.choice.id === memoryModels[row.provider] && !model.choice.audio,
   599→      );
   600→      if (!definition) continue;
   601→      if (otherConnections) {
   602→        if (row === current) return { row, definition };
   603→        const peers = candidates.filter(
   604→          (candidate) =>
   605→            candidate?.status === 'connected' &&
   606→            candidate.provider === row.provider &&
   607→            candidate.models.some(
   608→              (model) =>
   609→                model.choice.id === memoryModels[candidate.provider] && !model.choice.audio,
   610→            ),
   611→        );
   612→        if (peers.length === 1) return { row, definition };
   613→        continue;
   614→      }
   615→      const main = row.models.find((model) => model.choice.id === selected?.modelId);
   616→      if (!main) continue;
   617→      const price = modelPricing(definition);
   618→      const mainPrice = modelPricing(main);
   619→      if (
   620→        !price ||
   621→        price === 'variable' ||
   622→        price.input + price.output <= 0 ||
   623→        !mainPrice ||
   624→        mainPrice === 'variable' ||
   625→        mainPrice.input + mainPrice.output <= 0 ||
   626→        price.input > mainPrice.input ||
   627→        price.output > mainPrice.output
   628→      )
   629→        continue;
   630→      return { row, definition };
   631→    }
   632→  }

   723→  async function read(id: string) {
   724→    const result = await store.change(async (state) => {
   725→      const row = state.connections.find((item) => item.id === id);
   726→      if (!row) return {};
   727→      if (row.status === 'reconnect') throw attention(row);
   728→      try {
   729→        const current = credential(row);
   730→        if (current?.type !== 'oauth' || current.expires > Date.now() + 300000)
   731→          return { value: current };
   732→        const next = await backend.refresh(current);
   733→        remember(next);
   734→        row.credential = next;
   735→        return { value: next };
   736→      } catch (error) {
   737→        logFailure('model_refresh_failed', error instanceof Error ? (error.cause ?? error) : error);
   738→        if (!(error instanceof ModelAuthError) || !error.requiresReconnect)
   739→          throw new ConnectionError(
   740→            error instanceof ModelAuthError
   741→              ? error.message
   742→              : 'The model service is temporarily unavailable. Try again.',
   743→          );
   744→        row.status = 'reconnect';
   745→        return { reconnect: row };
   746→      }
   747→    });
   748→    if (result.reconnect) {
   749→      await reload();
   750→      throw attention(result.reconnect);
   751→    }
   752→    return result.value;
   753→  }

   809→  async function save(
   810→    provider: ModelProvider,
   811→    method: ModelMethod,
   812→    value: ModelCredential,
   813→    definitions: ModelDefinition[],
   814→    id?: string,
   815→    {
   816→      managed = false,
   817→      replace = false,
   818→      status = 'connected',
   819→      signal,
   820→    }: {
   821→      managed?: boolean;
   822→      replace?: boolean;
   823→      status?: ModelRecord['status'];
   824→      signal?: AbortSignal;
   825→    } = {},
   826→  ) {
   827→    remember(value);
   828→    if (!managed && !options.keyStore.configured)
   829→      throw new ConnectionError(
   830→        'Configure NOD_VAULT_KEY in server settings before saving a connection.',
   831→      );
   832→    if (status === 'connected' && !definitions.length && !modelProviderInfo[provider].speechOnly)
   833→      throw new ConnectionError('This connection has no supported models.');
   834→    const saved = await store.change((state) => {
   835→      const previous = state.connections.find((row) =>
   836→        id ? row.id === id : !managed && sameAccount(row, provider, method, value),
   837→      );
   838→      if (replace && !previous)
   839→        throw new ConnectionError('This model connection was removed. Start a new connection.');
   840→      const duplicates = managed
   841→        ? []
   842→        : state.connections.filter(
   843→            (row) => row !== previous && sameAccount(row, provider, method, value),
   844→          );
   845→      if (!previous && state.connections.length - duplicates.length >= 80)
   846→        throw new ConnectionError('Remove an unused model connection before adding another.');
   847→      const working = status === 'loading' && previous?.status === 'connected';
   848→      const row: ModelRecord = {
   849→        id: previous?.id ?? id ?? randomUUID(),
   850→        provider,
   851→        method,
   852→        label: connectionLabel(provider, method, value),
   853→        models: working ? previous.models : definitions,
   854→        managed,
   855→        status: working ? 'connected' : status,
   856→        requestFailures: previous?.requestFailures,
   857→        ...(!managed ? { credential: value } : {}),
   858→      };
   859→      state.connections = state.connections.flatMap((item) =>
   860→        item === previous ? [row] : duplicates.includes(item) ? [] : [item],
   861→      );
   862→      if (!previous) state.connections.push(row);
   863→      if (state.selection && duplicates.some((item) => item.id === state.selection!.connectionId))
   864→        state.selection = { ...state.selection, connectionId: row.id };
   865→      selectDefault(state, row);
   866→      return row.id;
   867→    }, signal);
   868→    await reload();
   869→    return saved;
   870→  }
   871→  function selectDefault(state: ModelState, row: ModelRecord) {
   872→    const { provider, method, models: definitions, id, managed, status } = row;
   873→    const choices = definitions.map((model) => model.choice);
   874→    const selected = state.selection;
   875→    const choice = choices.find((model) => model.id === selected?.modelId);
   876→    if (
   877→      selected?.connectionId === id &&
   878→      choice?.efforts.length &&
   879→      !choice.efforts.includes(selected.effort)
   880→    )
   881→      selected.effort = lightestEffort(choice.efforts);
   882→    for (const route of Object.values(state.routes)) {
   883→      if (!route || route === 'light' || route.connectionId !== id) continue;
   884→      const routed = choices.find((model) => model.id === route.modelId);
   885→      if (routed?.efforts.length && !routed.efforts.includes(route.effort))
   886→        route.effort = lightestEffort(routed.efforts);
   887→    }
   888→    const configuredModel =
   889→      managed && options.bootstrap?.provider === provider ? options.bootstrap.model : undefined;
   890→    const preferred = configuredModel
   891→      ? choices.find((model) => model.id === configuredModel)
   892→      : defaultModel(provider, choices, method);
   893→    if (
   894→      preferred &&
   895→      status === 'connected' &&
   896→      (!managed || options.bootstrap?.provider === provider) &&
   897→      (!state.selectionSet || (!managed && !selectedConnectionWorks(state)))
   898→    ) {
   899→      state.selection = {
   900→        connectionId: id,
   901→        modelId: preferred.id,
   902→        effort: defaultEffort(provider, preferred),
   903→      };
   904→      state.selectionSet = true;
   905→    }
   906→  }
   907→  async function updateCatalog(id: string, signal?: AbortSignal) {
   908→    signal?.throwIfAborted();
   909→    const snapshot = (await store.read()).connections.find((item) => item.id === id);
   910→    if (!snapshot) throw new ConnectionError('This model connection is unavailable.');
   911→    if (!snapshot.managed && snapshot.status === 'reconnect') throw attention(snapshot);
   912→    const value = credential(snapshot);
   913→    if (!value) throw new ConnectionError('This model connection is unavailable.');
   914→    const previous = JSON.stringify(snapshot);
   915→    const definitions = await catalog(snapshot.provider, snapshot.method, value, () =>
   916→      needsReconnect(id, snapshot),
   917→    );
   918→    if (!definitions.length && !modelProviderInfo[snapshot.provider].speechOnly)
   919→      throw new ConnectionError('This connection has no supported models.');
   920→    await store.change((state) => {
   921→      const row = state.connections.find((item) => item.id === id);
   922→      if (!row || JSON.stringify(row) !== previous)
   923→        throw new ConnectionError('This model connection changed. Refresh its models again.');
   924→      signal?.throwIfAborted();
   925→      row.models = definitions;
   926→      row.status = 'connected';
   927→      row.label = connectionLabel(row.provider, row.method, value);
   928→      selectDefault(state, row);
   929→    }, signal);
   930→    await reload();
   931→  }

   933→    const row = rows.find((row) => row.id === id);
   934→    if (!row) throw new ConnectionError('This model connection is unavailable.');
   935→    if (!row.managed) await read(id);
   936→    await updateCatalog(id);
   937→  }
   938→  function signInStatus(state: string) {

   947→  async function complete(callback: URL) {
   948→    const state = callback.searchParams.get('state') ?? '';
   949→    const attempt = attempts.get(state);
   950→    if (!attempt || attempt.status !== 'waiting')
   951→      throw new ConnectionError('This sign-in request has expired. Start a new connection.');
   952→    active(attempt);
   953→    attempt.status = 'completing';
   954→    if (attempt.provider === 'openai') {
   955→      try {
   956→        attempt.complete!(callback);
   957→      } catch (error) {
   958→        await cancel(state, Object(Object(error).cause).reason ?? 'callback_rejected');
   959→        throw error;
   960→      }
   961→    } else {
   962→      attempt.completion = finish(
   963→        attempt,
   964→        backend.exchange('openrouter', attempt.secret, callback),
   965→      );
   966→      attempt.completion.catch(() => {});
   967→    }
   968→    return attempt.completion;
   969→  }
   970→  async function finish(
   971→    attempt: NonNullable<ReturnType<typeof attempts.get>>,
   972→    result: Promise<ModelCredential>,
   973→  ) {
   974→    const method = attempt.provider === 'openai' ? 'subscription' : 'oauth';
   975→    try {
   976→      const value = await result;
   977→      active(attempt);
   978→      attempt.status = 'completing';
   979→      const id = await save(attempt.provider, method, value, [], attempt.connectionId, {
   980→        replace: !!attempt.connectionId,
   981→        status: 'loading',
   982→        signal: attempt.controller.signal,
   983→      });
   984→      attempt.connectionId = id;
   985→      active(attempt);
   986→      try {
   987→        await updateCatalog(id, attempt.controller.signal);
   988→      } catch (error) {
   989→        active(attempt);
   990→        attempt.error =
   991→          error instanceof ConnectionError
   992→            ? error.message
   993→            : 'Account connected. Refresh the connection to load its models.';
   994→      }
   995→      active(attempt);
   996→      attempt.status = 'connected';
   997→      connectionChanged(attempt.provider, method, 'connected');
   998→    } catch (error) {
   999→      const cause = error instanceof Error ? (error.cause ?? error) : error;
  1000→      logFailure('model_sign_in_failed', cause, signInFailure(attempt, cause));
  1001→      if (!attempt.controller.signal.aborted) connectionChanged(attempt.provider, method, 'failed');
  1002→      attempt.status = 'failed';
  1003→      attempt.error =
  1004→        error instanceof ConnectionError || error instanceof ModelAuthError
  1005→          ? error.message
  1006→          : 'Sign-in failed. Start a new connection.';
  1007→      throw error;
  1008→    } finally {
  1009→      attempt.secret = {};
  1010→    }
  1011→  }
  1012→  async function completeCallback(callback: URL) {
  1013→    const attempt = attempts.get(callback.searchParams.get('state') ?? '');
  1014→    await (attempt?.completion &&
  1015→    attempt.status !== 'waiting' &&
  1016→    !closed &&
  1017→    attempt.expires >= Date.now()
  1018→      ? attempt.completion
  1019→      : complete(callback));
  1020→    return attempt?.provider;
  1021→  }
  1022→  function start(provider: 'openai' | 'openrouter', connectionId?: string) {

  1027→  async function begin(provider: 'openai' | 'openrouter', connectionId?: string) {
  1028→    if (closed) throw new ConnectionError('Model connections have stopped.');
  1029→    try {
  1030→      await checkStorage();
  1031→    } catch (error) {
  1032→      connectionChanged(provider, provider === 'openai' ? 'subscription' : 'oauth', 'failed');
  1033→      throw error;
  1034→    }
  1035→    const saved = await store.read();
  1036→    await reload();
  1037→    const row = connectionId
  1038→      ? rows.find(
  1039→          (row) => row.id === connectionId && row.provider === provider && row.method !== 'api_key',
  1040→        )
  1041→      : undefined;
  1042→    if (connectionId && !row) throw new ConnectionError('This model connection is unavailable.');
  1043→    for (const [state, attempt] of attempts)
  1044→      if (
  1045→        attempt.expires < Date.now() ||
  1046→        (provider === 'openai' &&
  1047→          attempt.provider === provider &&
  1048→          (attempt.controller.signal.aborted || ['waiting', 'completing'].includes(attempt.status)))
  1049→      )
  1050→        await cancel(state, attempt.expires < Date.now() ? 'timeout' : 'superseded');
  1051→    if (closed) throw new ConnectionError('Model connections have stopped.');
  1052→    const controller = new AbortController();
  1053→    pendingControllers.add(controller);
  1054→    let flow: Awaited<ReturnType<ModelBackend['signIn']>> | undefined;
  1055→    const started = Date.now();
  1056→    let timedOut = false;
  1057→    const timeout = setTimeout(() => {
  1058→      timedOut = true;
  1059→      controller.abort();
  1060→    }, 15000);
  1061→    timeout.unref();
  1062→    try {
  1063→      const auth =
  1064→        provider === 'openai'
  1065→          ? await backend.signIn({
  1066→              signal: controller.signal,
  1067→              hostId: saved.hostId,
  1068→              previous: saved.connections.find((item) => item.id === row?.id)?.credential,
  1069→            })
  1070→          : backend.authorize({
  1071→              provider,
  1072→              redirectUri: `${options.origin}/api/models/oauth/callback`,
  1073→            });
  1074→      flow = 'result' in auth ? auth : undefined;
  1075→      flow?.result.catch(() => {});
  1076→      controller.signal.throwIfAborted();
  1077→      const timer = setTimeout(() => {
  1078→        void cancel(auth.state, 'timeout');
  1079→      }, 600000);
  1080→      timer.unref();
  1081→      const attempt: NonNullable<ReturnType<typeof attempts.get>> = {
  1082→        provider,
  1083→        state: auth.state,
  1084→        controller,
  1085→        complete: flow?.complete,
  1086→        connectionId,
  1087→        secret: 'secret' in auth ? auth.secret : {},
  1088→        expires: Date.now() + 600000,
  1089→        timer,
  1090→        status: 'waiting',
  1091→      };
  1092→      attempts.set(auth.state, attempt);
  1093→      if (flow) {
  1094→        attempt.completion = finish(attempt, flow.result);
  1095→        attempt.completion.catch(() => {});
  1096→      }
  1097→      return { url: auth.url, attempt: auth.state };
  1098→    } catch (error) {

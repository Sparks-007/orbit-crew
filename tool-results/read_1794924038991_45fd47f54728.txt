    35→function profileNotes(profile: Record<string, unknown>) {
    36→  return Object.entries(profile).flatMap(([field, value]) => {
    37→    if (typeof value !== 'string') return [];
    38→    const area = field === 'preferences' ? ('preferences' as const) : ('about' as const);
    39→    const label = area === 'about' ? `${field[0]!.toUpperCase()}${field.slice(1)}: ` : '';
    40→    const budget = 120 - label.length;
    41→    const notes: string[] = [];
    42→    let current = '';
    43→    const push = (unit: string) => {
    44→      if (current && current.length + unit.length + 1 > budget) {
    45→        notes.push(current);
    46→        current = '';
    47→      }
    48→      current = current ? `${current} ${unit}` : unit;
    49→    };
    50→    for (const sentence of value.split(/\n+|(?<=[.!?])\s+/)) {
    51→      const flat = sentence.replace(/[\s\p{Cc}]+/gu, ' ').trim();
    52→      if (flat.length <= budget) {
    53→        if (flat) push(flat);
    54→      } else
    55→        for (const word of flat.split(' '))
    56→          for (let start = 0; start < word.length; start += budget)
    57→            push(word.slice(start, start + budget));
    58→    }
    59→    if (current) notes.push(current);
    60→    return notes.map((text) => ({ area, text: label + text }));
    61→  });
    62→}
